# MT-32 — WebSocket Real-Time Status

## 1. Context
Enable real-time execution status updates via WebSocket so the frontend can display live progress as workflow executes.

## 2. Prerequisites
- MT-30 complete (parallel executor)
- FastAPI with WebSocket support

## 3. Goal
- WebSocket endpoint for execution status streaming
- Broadcast stage status changes (pending→running→completed/failed)
- Progress percentage calculation
- Connection management (multiple clients)
- Heartbeat/ping-pong for connection health

## 4. Implementation

### 4.1 `backend/app/services/websocket/execution_ws.py`

```python
"""WebSocket manager for execution status."""
from fastapi import WebSocket
from typing import Dict, Set
import json
import uuid

class ConnectionManager:
    def __init__(self):
        self.active_connections: Dict[str, Set[WebSocket]] = {}
    
    async def connect(self, websocket: WebSocket, execution_id: str):
        await websocket.accept()
        if execution_id not in self.active_connections:
            self.active_connections[execution_id] = set()
        self.active_connections[execution_id].add(websocket)
    
    def disconnect(self, websocket: WebSocket, execution_id: str):
        if execution_id in self.active_connections:
            self.active_connections[execution_id].discard(websocket)
    
    async def broadcast(self, execution_id: str, message: dict):
        if execution_id in self.active_connections:
            for connection in self.active_connections[execution_id]:
                try:
                    await connection.send_json(message)
                except:
                    pass

manager = ConnectionManager()
```

### 4.2 `backend/app/api/websocket.py`

```python
"""WebSocket API endpoints."""
from fastapi import APIRouter, WebSocket, WebSocketDisconnect

router = APIRouter()

@router.websocket("/ws/executions/{execution_id}")
async def websocket_execution_status(websocket: WebSocket, execution_id: str):
    from backend.app.services.websocket.execution_ws import manager
    
    await manager.connect(websocket, execution_id)
    try:
        while True:
            await websocket.receive_text()  # Keep connection alive
    except WebSocketDisconnect:
        manager.disconnect(websocket, execution_id)
```

## 5. Frontend Integration

```typescript
// frontend/src/hooks/useExecutionStatus.ts
export const useExecutionStatus = (executionId: string) => {
  const [status, setStatus] = useState({});
  
  useEffect(() => {
    const ws = new WebSocket(`ws://localhost:8000/ws/executions/${executionId}`);
    
    ws.onmessage = (event) => {
      const data = JSON.parse(event.data);
      setStatus(prev => ({
        ...prev,
        [data.stage_id]: data.status
      }));
    };
    
    return () => ws.close();
  }, [executionId]);
  
  return status;
};
```
