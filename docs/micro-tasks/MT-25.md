### 4.1 Create `backend/app/models/routing_rule.py`

```python
"""
Routing Rule and Decision models.
"""
import uuid
from datetime import datetime, timezone
from decimal import Decimal

from sqlalchemy import (
    Column, String, Integer, Boolean, DateTime,
    Numeric, ForeignKey, Index, UUID
)
from sqlalchemy.dialects.postgresql import JSONB
from sqlalchemy.orm import relationship

from backend.app.database import Base


class RoutingRule(Base):
    """
    Defines routing rules for stage types.
    """
    __tablename__ = "routing_rules"

    id = Column(UUID(as_uuid=True), primary_key=True, default=uuid.uuid4)
    stage_type = Column(String(100), nullable=False, unique=True)  # analysis, design, generation, etc.
    priority_factor = Column(String(50), default="balanced")  # cost, speed, quality, balanced
    
    # Constraints
    max_cost_per_call = Column(Numeric(10, 4), nullable=True)  # Maximum acceptable cost
    max_latency_ms = Column(Integer, nullable=True)  # Maximum acceptable latency
    min_capability_score = Column(Numeric(3, 2), default=0.80)  # Minimum capability score
    
    # Preferred models
    preferred_model_id = Column(UUID(as_uuid=True), ForeignKey("model_profiles.id", ondelete="SET NULL"), nullable=True)
    fallback_model_id = Column(UUID(as_uuid=True), ForeignKey("model_profiles.id", ondelete="SET NULL"), nullable=True)
    
    # Additional configuration
    config = Column(JSONB, default={})
    is_active = Column(Boolean, default=True)
    
    created_at = Column(DateTime(timezone=True), default=lambda: datetime.now(timezone.utc))
    updated_at = Column(DateTime(timezone=True), default=lambda: datetime.now(timezone.utc), onupdate=lambda: datetime.now(timezone.utc))
    
    # Relationships
    preferred_model = relationship("ModelProfile", foreign_keys=[preferred_model_id])
    fallback_model = relationship("ModelProfile", foreign_keys=[fallback_model_id])
    
    # Indexes
    __table_args__ = (
        Index("idx_routing_rules_stage_type", "stage_type"),
        Index("idx_routing_rules_is_active", "is_active"),
    )


class RoutingDecision(Base):
    """
    Logs routing decisions for analytics.
    """
    __tablename__ = "routing_decisions"

    id = Column(UUID(as_uuid=True), primary_key=True, default=uuid.uuid4)
    execution_id = Column(UUID(as_uuid=True), nullable=False, index=True)
    stage_id = Column(UUID(as_uuid=True), ForeignKey("stages.id", ondelete="CASCADE"), nullable=False)
    stage_type = Column(String(100), nullable=False)
    
    # Selected model
    selected_model_id = Column(UUID(as_uuid=True), ForeignKey("model_profiles.id", ondelete="SET NULL"), nullable=True)
    selected_provider = Column(String(50), nullable=False)
    selected_model_name = Column(String(100), nullable=False)
    
    # Decision reasoning
    selection_reason = Column(String(500), nullable=True)
    priority_factor = Column(String(50), nullable=True)
    alternatives_considered = Column(JSONB, default=[])  # List of alternative models
    
    # Overrides
    was_user_override = Column(Boolean, default=False)
    was_fallback = Column(Boolean, default=False)
    
    created_at = Column(DateTime(timezone=True), default=lambda: datetime.now(timezone.utc))
    
    # Relationships
    selected_model = relationship("ModelProfile")
    stage = relationship("Stage")
    
    # Indexes
    __table_args__ = (
        Index("idx_routing_decisions_execution_id", "execution_id"),
        Index("idx_routing_decisions_stage_type", "stage_type"),
    )
```

### 4.3 Create `backend/app/services/router/routing_engine.py`

```python
"""
Routing Engine.
Selects the optimal model for each stage based on rules and requirements.
"""
import uuid
from typing import Optional, List, Tuple
from decimal import Decimal

from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession

from backend.app.models.model_profile import ModelProfile
from backend.app.models.routing_rule import RoutingRule, RoutingDecision
from backend.app.models.stage import Stage
from backend.app.services.router.model_registry import ModelRegistry


class RoutingEngine:
    """Engine for intelligent model routing."""
    
    def __init__(self, db: AsyncSession):
        self.db = db
        self.registry = ModelRegistry(db)
    
    async def select_model_for_stage(
        self,
        stage: Stage,
        execution_id: uuid.UUID,
        user_override: Optional[str] = None  # model_name override
    ) -> Tuple[ModelProfile, str]:
        """
        Select the best model for a stage.
        
        Args:
            stage: The stage to route
            execution_id: Current execution ID
            user_override: Optional user-specified model
            
        Returns:
            (ModelProfile, selection_reason)
        """
        alternatives_considered = []
        
        # User override takes precedence
        if user_override:
            model = await self._get_model_by_name(user_override)
            if model:
                await self._log_decision(
                    execution_id, stage, model,
                    reason=f"User override: {user_override}",
                    was_user_override=True,
                    alternatives=alternatives_considered
                )
                return model, f"User override: {user_override}"
        
        # Stage-specific model preference
        if stage.model_preference:
            model = await self._get_model_by_name(stage.model_preference)
            if model:
                await self._log_decision(
                    execution_id, stage, model,
                    reason=f"Stage preference: {stage.model_preference}",
                    alternatives=alternatives_considered
                )
                return model, f"Stage preference: {stage.model_preference}"
        
        # Get routing rule for this stage type
        rule = await self._get_routing_rule(stage.stage_type)
        
        # If rule specifies preferred model, use it
        if rule and rule.preferred_model and rule.preferred_model.is_available:
            await self._log_decision(
                execution_id, stage, rule.preferred_model,
                reason=f"Routing rule preferred model for {stage.stage_type}",
                priority_factor=rule.priority_factor,
                alternatives=alternatives_considered
            )
            return rule.preferred_model, f"Routing rule: {rule.priority_factor}"
        
        # Apply routing algorithm
        candidate_models = await self._get_candidate_models(stage, rule)
        alternatives_considered = [
            {"model": m.model_name, "provider": m.provider, "score": m.get_capability_score(stage.stage_type)}
            for m in candidate_models
        ]
        
        if not candidate_models:
            # Use fallback
            if rule and rule.fallback_model:
                await self._log_decision(
                    execution_id, stage, rule.fallback_model,
                    reason="Fallback model - no suitable candidates",
                    was_fallback=True,
                    alternatives=alternatives_considered
                )
                return rule.fallback_model, "Fallback model"
            else:
                # Default fallback
                default = await self.registry.get_model("openai", "gpt-4o-mini")
                await self._log_decision(
                    execution_id, stage, default,
                    reason="Default fallback model",
                    was_fallback=True,
                    alternatives=alternatives_considered
                )
                return default, "Default fallback"
        
        # Select best model based on priority factor
        priority = rule.priority_factor if rule else "balanced"
        selected = self._apply_priority_selection(candidate_models, priority, stage.stage_type)
        
        reason = f"Routing: {priority} priority, score={selected.get_capability_score(stage.stage_type):.2f}"
        await self._log_decision(
            execution_id, stage, selected,
            reason=reason,
            priority_factor=priority,
            alternatives=alternatives_considered
        )
        
        return selected, reason
    
    async def _get_candidate_models(
        self,
        stage: Stage,
        rule: Optional[RoutingRule]
    ) -> List[ModelProfile]:
        """Get candidate models that meet stage requirements."""
        # Get all available models
        all_models = await self.registry.list_models(available_only=True)
        
        candidates = []
        min_score = float(rule.min_capability_score) if rule else 0.70
        
        for model in all_models:
            # Check capability score
            if model.get_capability_score(stage.stage_type) < min_score:
                continue
            
            # Check cost constraint
            if rule and rule.max_cost_per_call:
                # Estimate cost for typical 1K input + 500 output tokens
                estimated_cost = model.estimate_cost(1000, 500)
                if estimated_cost > rule.max_cost_per_call:
                    continue
            
            # Check latency constraint
            if rule and rule.max_latency_ms and model.avg_latency_ms:
                if model.avg_latency_ms > rule.max_latency_ms:
                    continue
            
            candidates.append(model)
        
        return candidates
    
    def _apply_priority_selection(
        self,
        candidates: List[ModelProfile],
        priority: str,
        stage_type: str
    ) -> ModelProfile:
        """Select model based on priority factor."""
        if priority == "cost":
            # Cheapest model
            return min(candidates, key=lambda m: (
                m.cost_per_input_token + m.cost_per_output_token
            ) / 2)
        
        elif priority == "speed":
            # Fastest model (lowest latency)
            with_latency = [m for m in candidates if m.avg_latency_ms is not None]
            if with_latency:
                return min(with_latency, key=lambda m: m.avg_latency_ms)
            return candidates[0]
        
        elif priority == "quality":
            # Highest capability score
            return max(candidates, key=lambda m: m.get_capability_score(stage_type))
        
        else:  # balanced
            # Balanced score: capability / (cost * latency_factor)
            def balanced_score(model: ModelProfile) -> float:
                capability = model.get_capability_score(stage_type)
                avg_cost = (model.cost_per_input_token + model.cost_per_output_token) / 2
                latency_factor = (model.avg_latency_ms or 1000) / 1000  # Normalize to seconds
                
                # Avoid division by zero
                if avg_cost == 0:
                    avg_cost = Decimal("0.0000000001")
                
                return capability / (float(avg_cost) * latency_factor * 1000000)  # Scale cost
            
            return max(candidates, key=balanced_score)
    
    async def _get_routing_rule(self, stage_type: str) -> Optional[RoutingRule]:
        """Get routing rule for stage type."""
        result = await self.db.execute(
            select(RoutingRule).where(
                RoutingRule.stage_type == stage_type,
                RoutingRule.is_active == True
            )
        )
        return result.scalar_one_or_none()
    
    async def _get_model_by_name(self, model_name: str) -> Optional[ModelProfile]:
        """Get model by name (checks all providers)."""
        result = await self.db.execute(
            select(ModelProfile).where(
                ModelProfile.model_name == model_name,
                ModelProfile.is_available == True
            )
        )
        return result.scalar_one_or_none()
    
    async def _log_decision(
        self,
        execution_id: uuid.UUID,
        stage: Stage,
        selected_model: ModelProfile,
        reason: str,
        priority_factor: Optional[str] = None,
        was_user_override: bool = False,
        was_fallback: bool = False,
        alternatives: List[dict] = None
    ):
        """Log routing decision for analytics."""
        decision = RoutingDecision(
            execution_id=execution_id,
            stage_id=stage.id,
            stage_type=stage.stage_type,
            selected_model_id=selected_model.id,
            selected_provider=selected_model.provider,
            selected_model_name=selected_model.model_name,
            selection_reason=reason,
            priority_factor=priority_factor,
            was_user_override=was_user_override,
            was_fallback=was_fallback,
            alternatives_considered=alternatives or []
        )
        self.db.add(decision)
        await self.db.flush()
```

"""
Routing Rules Management.
"""
import uuid
from typing import Optional, List

from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession

from backend.app.models.routing_rule import RoutingRule
from backend.app.models.model_profile import ModelProfile


class RoutingRulesManager:
    """Manage routing rules."""
    
    def __init__(self, db: AsyncSession):
        self.db = db
    
    async def create_rule(
        self,
        stage_type: str,
        priority_factor: str = "balanced",
        max_cost_per_call: Optional[float] = None,
        max_latency_ms: Optional[int] = None,
        min_capability_score: float = 0.80,
        preferred_model_id: Optional[uuid.UUID] = None,
        fallback_model_id: Optional[uuid.UUID] = None
    ) -> RoutingRule:
        """Create a new routing rule."""
        rule = RoutingRule(
            stage_type=stage_type,
            priority_factor=priority_factor,
            max_cost_per_call=max_cost_per_call,
            max_latency_ms=max_latency_ms,
            min_capability_score=min_capability_score,
            preferred_model_id=preferred_model_id,
            fallback_model_id=fallback_model_id
        )
        self.db.add(rule)
        await self.db.flush()
        return rule
    
    async def get_rule(self, stage_type: str) -> Optional[RoutingRule]:
        """Get routing rule for a stage type."""
        result = await self.db.execute(
            select(RoutingRule).where(RoutingRule.stage_type == stage_type)
        )
        return result.scalar_one_or_none()
    
    async def list_rules(self, active_only: bool = True) -> List[RoutingRule]:
        """List all routing rules."""
        query = select(RoutingRule)
        if active_only:
            query = query.where(RoutingRule.is_active == True)
        result = await self.db.execute(query)
        return result.scalars().all()
    
    async def update_rule(
        self,
        stage_type: str,
        **updates
    ) -> Optional[RoutingRule]:
        """Update a routing rule."""
        rule = await self.get_rule(stage_type)
        if not rule:
            return None
        
        for key, value in updates.items():
            setattr(rule, key, value)
        
        await self.db.flush()
        return rule
    
    async def delete_rule(self, stage_type: str) -> bool:
        """Delete a routing rule."""
        rule = await self.get_rule(stage_type)
        if not rule:
            return False
        
        await self.db.delete(rule)
        await self.db.flush()
        return True


async def seed_default_routing_rules(db: AsyncSession):
    """Seed default routing rules for common stage types."""
    rules_manager = RoutingRulesManager(db)
    
    # Get some model IDs for preferred/fallback

    gpt4o_mini = await registry.get_model("openai", "gpt-4o-mini")

    claude_haiku = await registry.get_model("anthropic", "claude-3-5-haiku-20241022")
    
    # Analysis: Balanced, prefer Claude Haiku for speed
    await rules_manager.create_rule(
