# MT-21 — Cache Integration & Management API

## 1. Context
With semantic caching fully implemented (MT-18, MT-19, MT-20), we now integrate it into the execution flow and create management API endpoints. When executing a workflow, the engine checks the cache before calling the LLM. We also add endpoints to view cache statistics, manage cache entries, and configure cache behavior.

## 2. Prerequisites
| What | Why |
|------|-----|
| **MT-20 complete** | Cache validation logic exists |
| **MT-14 complete** | Execution engine to integrate with |
| **MT-15 complete** | Execution API structure |

## 3. Goal
- Integrate semantic cache into execution engine (check cache before LLM call).
- `GET /api/cache/entries` — List all cache entries with filters.
- `GET /api/cache/stats` — Cache performance statistics.
- `DELETE /api/cache/entries/{id}` — Delete specific cache entry.
- `POST /api/cache/clear` — Clear cache by workflow/stage/type.
- `POST /api/cache/warm` — Pre-warm cache for common workflows.
- Cache hit/miss logging in execution records.
- Admin endpoint to force cache refresh.

```python
# Add imports
from backend.app.services.cache.semantic_cache import SemanticCacheService
from backend.app.config import settings

class ExecutionEngine:
    def __init__(self, db: AsyncSession):
        self.db = db
        self.context_mgr = ContextManager(db)
        self.cache_service = SemanticCacheService(db)  # Add cache service
    
    async def _execute_stage(self, workflow_id: uuid.UUID, execution_id: uuid.UUID,
                              stage: Stage, default_provider: str,
                              default_model: str | None) -> ExecutionRecord:
        """Execute a single stage with caching."""
        stage.status = "running"
        await self.db.flush()

        # Assemble input
        stage_input = await self.context_mgr.assemble_stage_input(workflow_id, execution_id, stage)

        # ========== CACHE LOOKUP ==========
        cache_result = await self.cache_service.lookup_with_validation(
            input_text=stage_input,
            workflow_id=workflow_id,
            execution_id=execution_id,
            current_stage=stage
        )

        if cache_result.cache_hit and cache_result.entry:
            # CACHE HIT - Return cached result
            print(f"[CACHE HIT] Stage {stage.name} - Similarity: {cache_result.similarity_score:.4f}")
            
            # Store cached result in context
            await self.context_mgr.add_context(
                workflow_id, execution_id, stage.id,
                content=cache_result.entry.result,
                context_type="stage_output",
                token_count=cache_result.entry.result_tokens,
            )

            # Record execution with cache hit flag
            record = ExecutionRecord(
                workflow_id=workflow_id, stage_id=stage.id, execution_id=execution_id,
                model_used=cache_result.entry.model_used,
                provider="cache",  # Special provider for cache hits
                input_tokens=0,  # No LLM call
                output_tokens=cache_result.entry.result_tokens,
                latency_ms=0,  # Instant from cache
                estimated_cost=0.0,  # No cost
                status="completed",
                result=cache_result.entry.result,
                started_at=datetime.now(timezone.utc),
                completed_at=datetime.now(timezone.utc),
            )
            self.db.add(record)
            stage.status = "completed"
            await self.db.flush()
            
            return record
        
        # ========== CACHE MISS - Execute with LLM ==========
        print(f"[CACHE MISS] Stage {stage.name} - Executing with LLM")

        # Determine provider and model
        provider_name = default_provider
        model = stage.model_preference or default_model

        # Get provider
        provider = registry.get(provider_name)
        if model is None:
            model = provider.get_default_model()

        # Execute with retry
        response = await self._call_with_retry(provider, stage_input, model)

        # Store result in context
        await self.context_mgr.add_context(
            workflow_id, execution_id, stage.id,
            content=response.content, context_type="stage_output",
            token_count=response.total_tokens,
        )

        # ========== STORE IN CACHE ==========
        ttl_seconds = settings.CACHE_TTL_SECONDS if settings.CACHE_TTL_SECONDS > 0 else None
        await self.cache_service.store_with_dependencies(
            input_text=stage_input,
            result=response.content,
            workflow_id=workflow_id,
            execution_id=execution_id,
            current_stage=stage,
            result_tokens=response.total_tokens,
            model_used=response.model,
            ttl_seconds=ttl_seconds
        )

        # Record execution
        record = ExecutionRecord(
            workflow_id=workflow_id, stage_id=stage.id, execution_id=execution_id,
            model_used=response.model, provider=response.provider,
            input_tokens=response.input_tokens, output_tokens=response.output_tokens,
            latency_ms=response.latency_ms, estimated_cost=response.estimated_cost,
            status="completed", result=response.content,
            started_at=datetime.now(timezone.utc),
            completed_at=datetime.now(timezone.utc),
        )
        self.db.add(record)

        # Update stage status
        stage.status = "completed"
        await self.db.flush()

        return record
```

```python
"""
Cache Management API endpoints.
"""
import uuid
from typing import List, Optional
from decimal import Decimal

from fastapi import APIRouter, Depends, HTTPException, Query
from sqlalchemy import select, func, and_
from sqlalchemy.ext.asyncio import AsyncSession

from backend.app.database import get_db
from backend.app.models.cache import CacheEntry
from backend.app.schemas.cache import CacheEntryResponse, CacheStats
from backend.app.services.cache.semantic_cache import SemanticCacheService
from backend.app.services.cache.cache_validator import CacheValidator


router = APIRouter(prefix="/api/cache", tags=["cache"])


@router.get("/entries", response_model=List[CacheEntryResponse])
async def list_cache_entries(
    workflow_id: Optional[uuid.UUID] = Query(None, description="Filter by workflow"),
    stage_type: Optional[str] = Query(None, description="Filter by stage type"),
    is_valid: Optional[bool] = Query(None, description="Filter by validity"),
    limit: int = Query(50, ge=1, le=200),
    offset: int = Query(0, ge=0),
    db: AsyncSession = Depends(get_db)
):
    """List cache entries with optional filters."""
    query = select(CacheEntry)
    
    if workflow_id:
        query = query.where(CacheEntry.workflow_id == workflow_id)
    if stage_type:
        query = query.where(CacheEntry.stage_type == stage_type)
    if is_valid is not None:
        query = query.where(CacheEntry.is_valid == is_valid)
    
    query = query.order_by(CacheEntry.created_at.desc()).limit(limit).offset(offset)
    
    result = await db.execute(query)
    entries = result.scalars().all()
    
    return [CacheEntryResponse.model_validate(e) for e in entries]


@router.get("/stats", response_model=CacheStats)
async def get_cache_stats(
    workflow_id: Optional[uuid.UUID] = Query(None, description="Filter by workflow"),
    db: AsyncSession = Depends(get_db)
):
    """Get cache performance statistics."""
    query = select(
        func.count(CacheEntry.id).label("total_entries"),
        func.sum(CacheEntry.hit_count).label("total_hits"),
        func.sum(CacheEntry.result_tokens).label("total_tokens"),
    ).where(CacheEntry.is_valid == True)
    
    if workflow_id:
        query = query.where(CacheEntry.workflow_id == workflow_id)
    
    result = await db.execute(query)
    row = result.first()
    
    total_entries = row.total_entries or 0
    total_hits = row.total_hits or 0
    total_tokens = row.total_tokens or 0
    
    # Calculate hit rate
    hit_rate = 0.0
    if total_entries > 0:
        # Estimate: total_hits / (total_hits + total_entries)
        # Assumes 1 initial store + N hits per entry
        total_lookups = total_entries + total_hits
        hit_rate = total_hits / total_lookups if total_lookups > 0 else 0.0
    
    # Estimate cost saved (simplified)
    cost_per_token = Decimal("0.000001")  # $0.001 per 1K tokens
    total_cost_saved = Decimal(total_tokens) * cost_per_token
    
    return CacheStats(
        total_entries=total_entries,
        total_hits=total_hits,
        hit_rate=hit_rate,
        total_tokens_saved=total_tokens,
        total_cost_saved=total_cost_saved,
        avg_similarity_score=0.95  # TODO: Calculate actual average
    )


@router.delete("/entries/{entry_id}")
async def delete_cache_entry(
    entry_id: uuid.UUID,
    db: AsyncSession = Depends(get_db)
):
    """Delete a specific cache entry."""
    result = await db.execute(
        select(CacheEntry).where(CacheEntry.id == entry_id)
    )
    entry = result.scalar_one_or_none()
    
    if not entry:
        raise HTTPException(status_code=404, detail=f"Cache entry {entry_id} not found")
    
    await db.delete(entry)
    await db.commit()
    
    return {"message": "Cache entry deleted", "id": entry_id}


@router.post("/clear")
async def clear_cache(
    workflow_id: Optional[uuid.UUID] = Query(None),
    stage_type: Optional[str] = Query(None),
    db: AsyncSession = Depends(get_db)
):
    """Clear cache entries by workflow, stage type, or all."""
    cache_service = SemanticCacheService(db)
    
    count = await cache_service.invalidate(
        workflow_id=workflow_id,
        stage_type=stage_type
    )
    await db.commit()
    
    return {
        "message": "Cache cleared",
        "entries_invalidated": count,
        "workflow_id": workflow_id,
        "stage_type": stage_type
    }


@router.post("/cleanup-expired")
async def cleanup_expired_cache(db: AsyncSession = Depends(get_db)):
    """Clean up expired cache entries."""
    validator = CacheValidator(db)
    count = await validator.cleanup_expired_entries()
    await db.commit()
    
    return {
        "message": "Expired cache entries cleaned up",
        "entries_cleaned": count
    }

@router.get("/hit-rate/{workflow_id}")
async def get_workflow_cache_hit_rate(
    workflow_id: uuid.UUID,
    db: AsyncSession = Depends(get_db)
):
    """Get cache hit rate for a specific workflow."""
    # Count cache entries
    result = await db.execute(
        select(func.count(CacheEntry.id), func.sum(CacheEntry.hit_count))
        .where(CacheEntry.workflow_id == workflow_id, CacheEntry.is_valid == True)
    )
    total_entries, total_hits = result.first()
    total_entries = total_entries or 0
    total_hits = total_hits or 0
    
    # Calculate hit rate
    hit_rate = 0.0
    if total_entries > 0:
        total_lookups = total_entries + total_hits
        hit_rate = total_hits / total_lookups if total_lookups > 0 else 0.0
    
    return {
        "workflow_id": workflow_id,
        "total_cache_entries": total_entries,
        "total_cache_hits": total_hits,
        "hit_rate": hit_rate,
        "hit_rate_percentage": f"{hit_rate * 100:.2f}%"
    }
```
