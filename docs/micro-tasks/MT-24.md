# MT-24 — Model Registry & Capability Profiles

## 1. Context
Phase 4 introduces **stage-aware routing**: intelligently selecting the best LLM for each stage based on capabilities, cost, latency, and requirements. This MT creates a model registry that catalogs available models with their profiles (strengths, weaknesses, pricing, context limits).

## 2. Prerequisites
| What | Why |
|------|-----|
| **MT-03 complete** | Alembic migrations exist |
| **MT-12, MT-13 complete** | LLM providers implemented |
| **MT-01 complete** | Database connection configured |

## 3. Goal
- `backend/app/models/model_profile.py` — ModelProfile ORM model.
- Alembic migration for `model_profiles` table.
- `backend/app/services/router/model_registry.py` — Model registration and lookup.
- Capability profiles for each model (analysis, design, code generation, reasoning).
- Pricing data (cost per input/output token).
- Performance metadata (avg latency, context window).
- Seed data for OpenAI, Anthropic, Gemini models.
- API endpoint to list available models.

```python
"""
Model Profile ORM model.
Stores metadata about available LLM models.
"""
import uuid
from datetime import datetime, timezone
from decimal import Decimal

from sqlalchemy import (
    Column, String, Integer, Boolean, DateTime, 
    Numeric, Index, UUID
)
from sqlalchemy.dialects.postgresql import JSONB

from backend.app.database import Base


class ModelProfile(Base):
    __tablename__ = "model_profiles"

    id = Column(UUID(as_uuid=True), primary_key=True, default=uuid.uuid4)
    provider = Column(String(50), nullable=False)  # openai, anthropic, google
    model_name = Column(String(100), nullable=False)  # gpt-4o-mini, claude-3-5-sonnet, etc.
    display_name = Column(String(100), nullable=False)
    
    # Capabilities (0.0 to 1.0 scores)
    capabilities = Column(JSONB, nullable=False, default={})
    # Example: {
    #   "analysis": 0.9,
    #   "design": 0.85,
    #   "code_generation": 0.95,
    #   "reasoning": 0.88,
    #   "creative_writing": 0.75
    # }
    
    # Context and limits
    max_context_tokens = Column(Integer, nullable=False)
    max_output_tokens = Column(Integer, nullable=True)
    
    # Pricing (per token)
    cost_per_input_token = Column(Numeric(12, 10), nullable=False)  # e.g., 0.0000001500
    cost_per_output_token = Column(Numeric(12, 10), nullable=False)
    
    # Performance
    avg_latency_ms = Column(Integer, nullable=True)  # Average response time
    
    # Availability
    is_available = Column(Boolean, default=True)
    supports_streaming = Column(Boolean, default=False)
    supports_function_calling = Column(Boolean, default=False)
    
    # Metadata
    description = Column(String(500), nullable=True)
    strengths = Column(JSONB, default=[])  # List of strengths
    limitations = Column(JSONB, default=[])  # List of limitations
    
    created_at = Column(DateTime(timezone=True), default=lambda: datetime.now(timezone.utc))
    updated_at = Column(DateTime(timezone=True), default=lambda: datetime.now(timezone.utc), onupdate=lambda: datetime.now(timezone.utc))
    
    # Indexes
    __table_args__ = (
        Index("idx_model_profiles_provider", "provider"),
        Index("idx_model_profiles_is_available", "is_available"),
        Index("idx_model_profiles_provider_model", "provider", "model_name", unique=True),
    )
    
    def __repr__(self):
        return f"<ModelProfile(provider={self.provider}, model={self.model_name})>"
    
    def get_capability_score(self, capability: str) -> float:
        """Get capability score for a specific task type."""
        return self.capabilities.get(capability, 0.5)  # Default to 0.5 if unknown
    
    def estimate_cost(self, input_tokens: int, output_tokens: int) -> Decimal:
        """Estimate cost for a given token count."""
        return (
            Decimal(input_tokens) * self.cost_per_input_token +
            Decimal(output_tokens) * self.cost_per_output_token
        )
```

```python
"""
Model Registry Service.
Manages available models and their capability profiles.
"""
import uuid
from typing import List, Optional, Dict
from decimal import Decimal

from sqlalchemy import select, and_
from sqlalchemy.ext.asyncio import AsyncSession

from backend.app.models.model_profile import ModelProfile


class ModelRegistry:
    """Service for managing model profiles."""
    
    def __init__(self, db: AsyncSession):
        self.db = db
    
    async def register_model(
        self,
        provider: str,
        model_name: str,
        display_name: str,
        capabilities: Dict[str, float],
        max_context_tokens: int,
        cost_per_input_token: Decimal,
        cost_per_output_token: Decimal,
        **kwargs
    ) -> ModelProfile:
        """Register a new model or update existing."""
        # Check if model already exists
        result = await self.db.execute(
            select(ModelProfile).where(
                and_(
                    ModelProfile.provider == provider,
                    ModelProfile.model_name == model_name
                )
            )
        )
        existing = result.scalar_one_or_none()
        
        if existing:
            # Update existing
            existing.display_name = display_name
            existing.capabilities = capabilities
            existing.max_context_tokens = max_context_tokens
            existing.cost_per_input_token = cost_per_input_token
            existing.cost_per_output_token = cost_per_output_token
            for key, value in kwargs.items():
                setattr(existing, key, value)
            await self.db.flush()
            return existing
        else:
            # Create new
            profile = ModelProfile(
                provider=provider,
                model_name=model_name,
                display_name=display_name,
                capabilities=capabilities,
                max_context_tokens=max_context_tokens,
                cost_per_input_token=cost_per_input_token,
                cost_per_output_token=cost_per_output_token,
                **kwargs
            )
            self.db.add(profile)
            await self.db.flush()
            return profile
    
    async def get_model(self, provider: str, model_name: str) -> Optional[ModelProfile]:
        """Get a specific model profile."""
        result = await self.db.execute(
            select(ModelProfile).where(
                and_(
                    ModelProfile.provider == provider,
                    ModelProfile.model_name == model_name
                )
            )
        )
        return result.scalar_one_or_none()
    
    async def list_models(
        self,
        provider: Optional[str] = None,
        available_only: bool = True
    ) -> List[ModelProfile]:
        """List all registered models."""
        query = select(ModelProfile)
        
        if provider:
            query = query.where(ModelProfile.provider == provider)
        if available_only:
            query = query.where(ModelProfile.is_available == True)
        
        query = query.order_by(ModelProfile.provider, ModelProfile.model_name)
        
        result = await self.db.execute(query)
        return result.scalars().all()
    
    async def get_models_for_capability(
        self,
        capability: str,
        min_score: float = 0.7
    ) -> List[ModelProfile]:
        """Get models that excel at a specific capability."""
        all_models = await self.list_models(available_only=True)
        return [
            model for model in all_models
            if model.get_capability_score(capability) >= min_score
        ]
    
    async def get_cheapest_model(
        self,
        capability: Optional[str] = None,
        min_score: float = 0.7
    ) -> Optional[ModelProfile]:
        """Get the cheapest model that meets capability requirements."""
        if capability:
            models = await self.get_models_for_capability(capability, min_score)
        else:
            models = await self.list_models(available_only=True)
        
        if not models:
            return None
        
        # Sort by average cost (input + output)
        return min(
            models,
            key=lambda m: (m.cost_per_input_token + m.cost_per_output_token) / 2
        )
    
    async def get_fastest_model(
        self,
        capability: Optional[str] = None,
        min_score: float = 0.7
    ) -> Optional[ModelProfile]:
        """Get the fastest model that meets capability requirements."""
        if capability:
            models = await self.get_models_for_capability(capability, min_score)
        else:
            models = await self.list_models(available_only=True)
        
        models_with_latency = [m for m in models if m.avg_latency_ms is not None]
        if not models_with_latency:
            return None
        
        return min(models_with_latency, key=lambda m: m.avg_latency_ms)
```

### 4.4 Create `backend/app/schemas/model_profile.py`

```python
"""
Pydantic schemas for model profiles.
"""
import uuid
from datetime import datetime
from decimal import Decimal
from typing import Dict, List, Optional

from pydantic import BaseModel, Field


class ModelProfileResponse(BaseModel):
    """Response schema for model profile."""
    id: uuid.UUID
    provider: str
    model_name: str
    display_name: str
    capabilities: Dict[str, float]
    max_context_tokens: int
    max_output_tokens: Optional[int]
    cost_per_input_token: Decimal
    cost_per_output_token: Decimal
    avg_latency_ms: Optional[int]
    is_available: bool
    supports_streaming: bool
    supports_function_calling: bool
    description: Optional[str]
    strengths: List[str]
    limitations: List[str]
    created_at: datetime
    
    class Config:
        from_attributes = True


class ModelComparisonResponse(BaseModel):
    """Comparison of multiple models."""
    models: List[ModelProfileResponse]
    cheapest: Optional[str] = None  # model_name
    fastest: Optional[str] = None  # model_name
    most_capable: Optional[Dict[str, str]] = None  # capability -> model_name
```

### 4.5 Create `backend/app/api/models.py`

```python
"""
Model Registry API endpoints.
"""
from typing import List, Optional

from fastapi import APIRouter, Depends, Query
from sqlalchemy.ext.asyncio import AsyncSession

from backend.app.database import get_db
from backend.app.schemas.model_profile import ModelProfileResponse, ModelComparisonResponse
from backend.app.services.router.model_registry import ModelRegistry


router = APIRouter(prefix="/api/models", tags=["models"])


@router.get("", response_model=List[ModelProfileResponse])
async def list_models(
    provider: Optional[str] = Query(None, description="Filter by provider"),
    capability: Optional[str] = Query(None, description="Filter by capability"),
    min_score: float = Query(0.7, ge=0.0, le=1.0, description="Minimum capability score"),
    db: AsyncSession = Depends(get_db)
):
    """List all available models, optionally filtered."""
    registry = ModelRegistry(db)
    
    if capability:
        models = await registry.get_models_for_capability(capability, min_score)
    else:
        models = await registry.list_models(provider=provider)
    
    return [ModelProfileResponse.model_validate(m) for m in models]


@router.get("/compare", response_model=ModelComparisonResponse)
async def compare_models(
    capability: Optional[str] = Query(None),
    db: AsyncSession = Depends(get_db)
):
    """Compare all models and find best options."""
    registry = ModelRegistry(db)
    
    all_models = await registry.list_models()
    cheapest = await registry.get_cheapest_model(capability)
    fastest = await registry.get_fastest_model(capability)
    
    # Find most capable for common task types
    capabilities_to_check = ["analysis", "design", "code_generation", "reasoning"]
    most_capable = {}
    for cap in capabilities_to_check:
        models_for_cap = await registry.get_models_for_capability(cap, min_score=0.0)
        if models_for_cap:
            best = max(models_for_cap, key=lambda m: m.get_capability_score(cap))
            most_capable[cap] = best.model_name
    
    return ModelComparisonResponse(
        models=[ModelProfileResponse.model_validate(m) for m in all_models],
        cheapest=cheapest.model_name if cheapest else None,
        fastest=fastest.model_name if fastest else None,
        most_capable=most_capable
    )


@router.get("/{provider}/{model_name}", response_model=ModelProfileResponse)
async def get_model(
    provider: str,
    model_name: str,
    db: AsyncSession = Depends(get_db)
):
    """Get details for a specific model."""
    registry = ModelRegistry(db)
    model = await registry.get_model(provider, model_name)
    
    if not model:
        from fastapi import HTTPException
        raise HTTPException(status_code=404, detail=f"Model {provider}/{model_name} not found")
    
    return ModelProfileResponse.model_validate(model)
```
