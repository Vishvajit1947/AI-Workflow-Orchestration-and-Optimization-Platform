### 4.1 Create `frontend/src/api/analytics.ts`

```typescript
/**
 * Analytics API Client
 */
import { apiClient } from './client';

export interface WorkflowMetrics {
  total_executions: number;
  total_stages: number;
  total_tokens: number;
  total_cost: number;
  avg_latency_ms: number;
  success_rate: number;
  failure_rate: number;
}

export interface CacheMetrics {
  total_entries: number;
  total_hits: number;
  hit_rate: number;
  total_tokens_saved: number;
  total_cost_saved: number;
}

export interface ModelUtilization {
  provider: string;
  model: string;
  usage_count: number;
  total_tokens: number;
  total_cost: number;
  avg_latency_ms: number;
}

export interface CostTrendPoint {
  date: string;
  cost: number;
  executions: number;
}

export interface TokenTrendPoint {
  date: string;
  input_tokens: number;
  output_tokens: number;
  total_tokens: number;
}

export const analyticsApi = {
  /**
   * Get overview metrics
   */
  async getOverview(params?: {
    start_date?: string;
    end_date?: string;
  }): Promise<WorkflowMetrics> {
    const response = await apiClient.get('/analytics/overview', { params });
    return response.data;
  },

  /**
   * Get cache metrics
   */
  async getCacheMetrics(): Promise<CacheMetrics> {
    const response = await apiClient.get('/analytics/cache');
    return response.data;
  },

  /**
   * Get model utilization
   */
  async getModelUtilization(params?: {
    start_date?: string;
    end_date?: string;
  }): Promise<ModelUtilization[]> {
    const response = await apiClient.get('/analytics/models', { params });
    return response.data;
  },

  /**
   * Get cost trend
   */
  async getCostTrend(days: number = 30): Promise<CostTrendPoint[]> {
    const response = await apiClient.get('/analytics/costs', {
      params: { days }
    });
    return response.data;
  },

  /**
   * Get token trend
   */
  async getTokenTrend(days: number = 30): Promise<TokenTrendPoint[]> {
    const response = await apiClient.get('/analytics/tokens', {
      params: { days }
    });
    return response.data;
  },

  /**
   * Export analytics data
   */
  async exportData(format: 'csv' | 'json'): Promise<Blob> {
    const response = await apiClient.get('/analytics/export', {
      params: { format },
      responseType: 'blob'
    });
    return response.data;
  }
};
```

### 4.2 Create `frontend/src/components/metrics/MetricCard.tsx`

```typescript
/**
 * MetricCard Component
 * Displays a single metric with icon, label, and formatted value
 */
import React from 'react';

export interface MetricCardProps {
  icon: React.ReactNode;
  label: string;
  value: string | number;
  subtitle?: string;
  trend?: {
    value: number;
    direction: 'up' | 'down';
    label: string;
  };
  color?: 'blue' | 'green' | 'purple' | 'orange' | 'red';
}

export const MetricCard: React.FC<MetricCardProps> = ({
  icon,
  label,
  value,
  subtitle,
  trend,
  color = 'blue'
}) => {
  const colorClasses = {
    blue: 'bg-blue-50 text-blue-600',
    green: 'bg-green-50 text-green-600',
    purple: 'bg-purple-50 text-purple-600',
    orange: 'bg-orange-50 text-orange-600',
    red: 'bg-red-50 text-red-600'
  };

  const trendColorClasses = {
    up: 'text-green-600',
    down: 'text-red-600'
  };

  return (
    <div className="bg-white rounded-lg shadow p-6 hover:shadow-md transition-shadow">
      <div className="flex items-center justify-between mb-4">
        <div className={`p-3 rounded-lg ${colorClasses[color]}`}>
          {icon}
        </div>
        {trend && (
          <div className={`text-sm font-medium ${trendColorClasses[trend.direction]}`}>
            {trend.direction === 'up' ? '↑' : '↓'} {trend.value}%
          </div>
        )}
      </div>
      
      <div>
        <p className="text-sm text-gray-600 mb-1">{label}</p>
        <p className="text-3xl font-bold text-gray-900">{value}</p>
        {subtitle && (
          <p className="text-sm text-gray-500 mt-1">{subtitle}</p>
        )}
      </div>
    </div>
  );
};
```

### 4.4 Create `frontend/src/pages/Dashboard.tsx`

```typescript
/**
 * Dashboard Page
 * Main analytics dashboard with overview metrics
 */
import React, { useEffect, useState } from 'react';
import { MetricCard } from '../components/metrics/MetricCard';
import { MetricGrid } from '../components/metrics/MetricGrid';
import { analyticsApi, WorkflowMetrics, CacheMetrics } from '../api/analytics';

// Simple icons (replace with react-icons or heroicons if preferred)
const ActivityIcon = () => <div className="text-2xl">📊</div>;
const DollarIcon = () => <div className="text-2xl">💰</div>;
const TokenIcon = () => <div className="text-2xl">🎯</div>;
const ClockIcon = () => <div className="text-2xl">⏱️</div>;
const CacheIcon = () => <div className="text-2xl">💾</div>;
const SuccessIcon = () => <div className="text-2xl">✅</div>;

export const Dashboard: React.FC = () => {
  const [workflowMetrics, setWorkflowMetrics] = useState<WorkflowMetrics | null>(null);
  const [cacheMetrics, setCacheMetrics] = useState<CacheMetrics | null>(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    loadMetrics();
  }, []);

  const loadMetrics = async () => {
    try {
      setLoading(true);
      setError(null);
      
      const [workflow, cache] = await Promise.all([
        analyticsApi.getOverview(),
        analyticsApi.getCacheMetrics()
      ]);
      
      setWorkflowMetrics(workflow);
      setCacheMetrics(cache);
    } catch (err) {
      console.error('Failed to load metrics:', err);
      setError('Failed to load analytics data');
    } finally {
      setLoading(false);
    }
  };

  const formatCost = (cost: number) => {
    return `$${cost.toFixed(4)}`;
  };

  const formatNumber = (num: number) => {
    if (num >= 1000000) {
      return `${(num / 1000000).toFixed(1)}M`;
    }
    if (num >= 1000) {
      return `${(num / 1000).toFixed(1)}K`;
    }
    return num.toFixed(0);
  };

  const formatLatency = (ms: number) => {
    if (ms >= 1000) {
      return `${(ms / 1000).toFixed(2)}s`;
    }
    return `${ms.toFixed(0)}ms`;
  };

  if (loading) {
    return (
      <div className="flex items-center justify-center h-64">
        <div className="text-lg text-gray-600">Loading analytics...</div>
      </div>
    );
  }

  if (error) {
    return (
      <div className="bg-red-50 border border-red-200 rounded-lg p-4">
        <p className="text-red-800">{error}</p>
        <button
          onClick={loadMetrics}
          className="mt-2 text-red-600 hover:text-red-800 underline"
        >
          Retry
        </button>
      </div>
    );
  }

  if (!workflowMetrics || !cacheMetrics) {
    return (
      <div className="text-center text-gray-600">
        No analytics data available yet. Execute some workflows to see metrics.
      </div>
    );
  }

  return (
    <div className="space-y-8">
      {/* Header */}
      <div className="flex items-center justify-between">
        <div>
          <h1 className="text-3xl font-bold text-gray-900">Analytics Dashboard</h1>
          <p className="text-gray-600 mt-1">
            Platform performance and usage metrics
          </p>
        </div>
        <button
          onClick={() => analyticsApi.exportData('csv')}
          className="px-4 py-2 bg-blue-600 text-white rounded-lg hover:bg-blue-700 transition"
        >
          Export Data
        </button>
      </div>

      {/* Key Metrics Grid */}
      <MetricGrid columns={4}>
        <MetricCard
          icon={<ActivityIcon />}
          label="Total Executions"
          value={workflowMetrics.total_executions}
          subtitle={`${workflowMetrics.total_stages} stages`}
          color="blue"
        />
        
        <MetricCard
          icon={<DollarIcon />}
          label="Total Cost"
          value={formatCost(workflowMetrics.total_cost)}
          subtitle="All time"
          color="green"
        />
        
        <MetricCard
          icon={<TokenIcon />}
          label="Total Tokens"
          value={formatNumber(workflowMetrics.total_tokens)}
          subtitle="Input + Output"
          color="purple"
        />
        
        <MetricCard
          icon={<ClockIcon />}
          label="Avg Latency"
          value={formatLatency(workflowMetrics.avg_latency_ms)}
          subtitle="Per stage"
          color="orange"
        />
      </MetricGrid>

      {/* Performance Metrics */}
      <MetricGrid columns={3}>
        <MetricCard
          icon={<SuccessIcon />}
          label="Success Rate"
          value={`${workflowMetrics.success_rate.toFixed(1)}%`}
          subtitle={`${(workflowMetrics.failure_rate).toFixed(1)}% failures`}
          color="green"
        />
        
        <MetricCard
          icon={<CacheIcon />}
          label="Cache Hit Rate"
          value={`${cacheMetrics.hit_rate.toFixed(1)}%`}
          subtitle={`${cacheMetrics.total_hits} hits / ${cacheMetrics.total_entries} entries`}
          color="blue"
        />
        
        <MetricCard
          icon={<DollarIcon />}
          label="Cost Saved (Cache)"
          value={formatCost(cacheMetrics.total_cost_saved)}
          subtitle={`${formatNumber(cacheMetrics.total_tokens_saved)} tokens saved`}
          color="green"
        />
      </MetricGrid>

      {/* Additional sections will be added in MT-39 */}
    </div>
  );
};
```
