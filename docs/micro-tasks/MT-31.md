# MT-31 — Enhanced Error Handling

## 1. Context
With parallel execution (MT-30), we need robust error handling for production use: automatic retries with exponential backoff, model fallback when primary fails, circuit breakers for unhealthy services, and graceful degradation.

## 2. Prerequisites
| What | Why |
|------|-----|
| **MT-30 complete** | Parallel executor exists |
| **MT-25 complete** | Routing rules with fallbacks |
| **MT-14 complete** | Basic retry mechanism |

## 3. Goal
- Exponential backoff for retries (avoid overwhelming failing services).
- Model fallback (switch to backup model on repeated failures).
- Circuit breaker pattern (stop calling failing provider temporarily).
- Timeout handling per stage.
- Rate limit detection and backoff.
- Error classification (transient vs permanent).
- Detailed error logging and reporting.

### 4.1 Create `backend/app/services/execution/error_handler.py`

```python
"""
Enhanced Error Handling.
Retries, backoff, fallbacks, circuit breakers.
"""
import asyncio
import time
from typing import Optional, Callable, Any
from datetime import datetime, timedelta
from collections import defaultdict

from backend.app.config import settings


class CircuitBreaker:
    """Circuit breaker for provider health tracking."""
    
    def __init__(self, failure_threshold: int = 5, timeout: int = 60):
        self.failure_threshold = failure_threshold
        self.timeout = timeout  # seconds
        self.failures: dict[str, int] = defaultdict(int)
        self.last_failure_time: dict[str, datetime] = {}
        self.circuit_open: dict[str, bool] = defaultdict(bool)
    
    def record_failure(self, provider: str):
        """Record a failure for a provider."""
        self.failures[provider] += 1
        self.last_failure_time[provider] = datetime.now()
        
        if self.failures[provider] >= self.failure_threshold:
            self.circuit_open[provider] = True
            print(f"[CIRCUIT BREAKER] Opened for {provider} (failures: {self.failures[provider]})")
    
    def record_success(self, provider: str):
        """Record a success (resets failure count)."""
        self.failures[provider] = 0
        self.circuit_open[provider] = False
    
    def is_open(self, provider: str) -> bool:
        """Check if circuit is open (provider unhealthy)."""
        if not self.circuit_open.get(provider, False):
            return False
        
        # Check if timeout has passed (half-open state)
        last_failure = self.last_failure_time.get(provider)
        if last_failure:
            elapsed = (datetime.now() - last_failure).total_seconds()
            if elapsed > self.timeout:
                print(f"[CIRCUIT BREAKER] Half-open for {provider} (testing)")
                self.circuit_open[provider] = False
                return False
        
        return True


class ErrorHandler:
    """Handles errors with retries, backoff, and fallbacks."""
    
    def __init__(self):
        self.circuit_breaker = CircuitBreaker()
        self.max_retries = settings.MAX_RETRIES
        self.backoff_base = settings.RETRY_BACKOFF_BASE
    
    async def execute_with_retry(
        self,
        func: Callable,
        *args,
        provider: str = "unknown",
        fallback_func: Optional[Callable] = None,
        max_retries: Optional[int] = None,
        **kwargs
    ) -> Any:
        """
        Execute function with retry, backoff, and fallback.
        
        Args:
            func: Function to execute
            provider: Provider name (for circuit breaker)
            fallback_func: Fallback function if all retries fail
            max_retries: Override default max retries
            *args, **kwargs: Arguments for func
            
        Returns:
            Result from func or fallback_func
            
        Raises:
            Last exception if all retries and fallback fail
        """
        retries = max_retries or self.max_retries
        last_error = None
        
        # Check circuit breaker
        if self.circuit_breaker.is_open(provider):
            print(f"[ERROR HANDLER] Circuit open for {provider}, trying fallback")
            if fallback_func:
                try:
                    result = await fallback_func(*args, **kwargs)
                    return result
                except Exception as e:
                    raise Exception(f"Circuit open for {provider} and fallback failed: {e}")
            else:
                raise Exception(f"Circuit open for {provider} and no fallback available")
        
        for attempt in range(retries):
            try:
                result = await func(*args, **kwargs)
                
                # Success - record for circuit breaker
                self.circuit_breaker.record_success(provider)
                
                if attempt > 0:
                    print(f"[RETRY SUCCESS] Succeeded on attempt {attempt + 1}")
                
                return result
            
            except Exception as e:
                last_error = e
                error_type = self._classify_error(e)
                
                print(f"[ERROR] Attempt {attempt + 1}/{retries} failed: {type(e).__name__}: {e}")
                
                # Record failure for circuit breaker
                self.circuit_breaker.record_failure(provider)
                
                # Check if error is retryable
                if error_type == "permanent":
                    print(f"[ERROR] Permanent error detected, not retrying")
                    break
                
                # Last attempt or permanent error
                if attempt >= retries - 1:
                    break
                
                # Exponential backoff
                if error_type == "rate_limit":
                    wait = 60  # 1 minute for rate limits
                else:
                    wait = self.backoff_base ** attempt
                
                print(f"[RETRY] Waiting {wait}s before retry...")
                await asyncio.sleep(wait)
        
        # All retries failed, try fallback
        if fallback_func:
            print(f"[FALLBACK] All retries failed, trying fallback")
            try:
                result = await fallback_func(*args, **kwargs)
                return result
            except Exception as fallback_error:
                print(f"[FALLBACK FAILED] {fallback_error}")
                raise Exception(f"All retries and fallback failed. Last error: {last_error}")
        
        # No fallback, raise last error
        raise last_error
    
    def _classify_error(self, error: Exception) -> str:
        """
        Classify error as transient, rate_limit, or permanent.
        
        Returns:
            "transient" - retry may succeed
            "rate_limit" - need longer backoff
            "permanent" - don't retry
        """
        error_str = str(error).lower()
        
        # Rate limit errors
        if "rate limit" in error_str or "429" in error_str or "quota" in error_str:
            return "rate_limit"
        
        # Transient errors (network, timeout, server errors)
        transient_indicators = [
            "timeout", "timed out", "connection", "network",
            "503", "502", "500", "server error", "unavailable"
        ]
        if any(ind in error_str for ind in transient_indicators):
            return "transient"
        
        # Permanent errors (bad request, auth, not found)
        permanent_indicators = [
            "400", "401", "403", "404", "invalid", "unauthorized",
            "authentication", "not found", "bad request"
        ]
        if any(ind in error_str for ind in permanent_indicators):
            return "permanent"
        
        # Default to transient (be optimistic)
        return "transient"
    
    async def execute_with_timeout(
        self,
        func: Callable,
        timeout_seconds: int,
        *args,
        **kwargs
    ) -> Any:
        """Execute function with timeout."""
        try:
            result = await asyncio.wait_for(
                func(*args, **kwargs),
                timeout=timeout_seconds
            )
            return result
        except asyncio.TimeoutError:
            raise TimeoutError(f"Operation timed out after {timeout_seconds}s")
```

"""
Error Handler Tests.
"""
import pytest
import asyncio
from backend.app.services.execution.error_handler import ErrorHandler, CircuitBreaker


@pytest.mark.asyncio
async def test_retry_with_exponential_backoff():
    """Test exponential backoff on retries."""
    import time
    
    handler = ErrorHandler()
    attempts = []
    
    async def failing_func():
        attempts.append(time.time())
        if len(attempts) < 3:
            raise Exception("Transient error")
        return "success"
    
    result = await handler.execute_with_retry(failing_func, max_retries=3)
    
    assert result == "success"
    assert len(attempts) == 3
    
    # Check backoff (2^0, 2^1, 2^2 seconds)
    if len(attempts) >= 3:
        gap1 = attempts[1] - attempts[0]
        gap2 = attempts[2] - attempts[1]
        assert gap1 >= 1.0  # At least 1 second
        assert gap2 >= 2.0  # At least 2 seconds


@pytest.mark.asyncio
async def test_fallback_on_all_retries_failed():
    """Test fallback when all retries fail."""
    handler = ErrorHandler()
    
    async def always_fails():
        raise Exception("Primary always fails")
    
    async def fallback():
        return "fallback_result"
    
    result = await handler.execute_with_retry(
        always_fails,
        fallback_func=fallback,
        max_retries=2
    )
    
    assert result == "fallback_result"


@pytest.mark.asyncio
async def test_circuit_breaker_opens():
    """Test circuit breaker opens after threshold."""
    breaker = CircuitBreaker(failure_threshold=3, timeout=60)
    
    # Record 3 failures
    for _ in range(3):
        breaker.record_failure("openai")
    
    assert breaker.is_open("openai") is True
    
    # Other providers not affected
    assert breaker.is_open("anthropic") is False


@pytest.mark.asyncio
async def test_circuit_breaker_half_open():
    """Test circuit breaker half-open after timeout."""
    breaker = CircuitBreaker(failure_threshold=2, timeout=1)
    
    # Open circuit
    breaker.record_failure("openai")
    breaker.record_failure("openai")
    assert breaker.is_open("openai") is True
    
    # Wait for timeout
    await asyncio.sleep(1.1)
    
    # Should be half-open (testing allowed)
    assert breaker.is_open("openai") is False


@pytest.mark.asyncio
async def test_timeout_handling():
    """Test operation timeout."""
    handler = ErrorHandler()
    
    async def slow_func():
        await asyncio.sleep(5)
        return "too late"
    
    with pytest.raises(TimeoutError):
        await handler.execute_with_timeout(slow_func, timeout_seconds=1)


@pytest.mark.asyncio
async def test_rate_limit_classification():
    """Test rate limit error classification."""
    handler = ErrorHandler()
    
    error = Exception("Rate limit exceeded (429)")
    error_type = handler._classify_error(error)
    
    assert error_type == "rate_limit"


@pytest.mark.asyncio
async def test_permanent_error_no_retry():
    """Test that permanent errors are not retried."""
    handler = ErrorHandler()
    attempts = []
    
    async def bad_request():
        attempts.append(1)
        raise Exception("400 Bad Request: Invalid input")
    
    with pytest.raises(Exception):
        await handler.execute_with_retry(bad_request, max_retries=3)
    
    # Should only attempt once (no retries for permanent errors)
    assert len(attempts) == 1
