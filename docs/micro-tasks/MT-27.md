# MT-27 — Frontend Routing Config UI

## 1. Context
With routing integrated into execution (MT-26), we need frontend UI to visualize routing decisions, configure routing rules, and allow users to override model selection per stage. This makes routing transparent and configurable.

## 2. Prerequisites
| What | Why |
|------|-----|
| **MT-26 complete** | Routing in execution engine |
| **MT-24 complete** | Model registry API available |
| **MT-16 complete** | Execution UI exists |

## 3. Goal
- Model selection dropdown on stage editor.
- Routing rules configuration page.
- Routing decision visualization in execution results.
- Model comparison tool.
- Override model selection in execution modal.
- Show which model was used per stage in results.

### 4.2 Create `frontend/src/api/modelsApi.ts`

```typescript
/**
 * API client for model operations.
 */
import { apiClient } from './client';
import type { ModelProfile } from '../types/routing';

export const modelsApi = {
  /**
   * List all available models.
   */
  async listModels(params?: {
    provider?: string;
    capability?: string;
    min_score?: number;
  }): Promise<ModelProfile[]> {
    const response = await apiClient.get('/api/models', { params });
    return response.data;
  },

  /**
   * Get specific model details.
   */
  async getModel(provider: string, modelName: string): Promise<ModelProfile> {
    const response = await apiClient.get(`/api/models/${provider}/${modelName}`);
    return response.data;
  },

  /**
   * Compare models.
   */
  async compareModels(capability?: string): Promise<{
    models: ModelProfile[];
    cheapest: string | null;
    fastest: string | null;
    most_capable: Record<string, string>;
  }> {
    const params = capability ? { capability } : {};
    const response = await apiClient.get('/api/models/compare', { params });
    return response.data;
  }
};
```

```typescript
import { useState, useEffect } from 'react';
import { modelsApi } from '../api/modelsApi';
import type { ModelProfile } from '../types/routing';

// Inside StageEditor component:
const [availableModels, setAvailableModels] = useState<ModelProfile[]>([]);

useEffect(() => {
  loadModels();
}, [stage?.stage_type]);

const loadModels = async () => {
  try {
    const models = await modelsApi.listModels({
      capability: stage?.stage_type,
      min_score: 0.7
    });
    setAvailableModels(models);
  } catch (error) {
    console.error('Failed to load models:', error);
  }
};

// Add to form:
<div className="mb-4">
  <label className="block text-sm font-medium text-gray-700 mb-2">
    Model Preference (Optional)
  </label>
  <select
    value={stage.model_preference || ''}
    onChange={(e) => setStage({ ...stage, model_preference: e.target.value || null })}
    className="w-full px-3 py-2 border border-gray-300 rounded-md focus:outline-none focus:ring-2 focus:ring-blue-500"
  >
    <option value="">Auto (Use routing rules)</option>
    {availableModels.map((model) => (
      <option key={model.id} value={model.model_name}>
        {model.display_name} - {model.provider} 
        {model.capabilities[stage.stage_type] && 
          ` (${(model.capabilities[stage.stage_type] * 100).toFixed(0)}% match)`
        }
      </option>
    ))}
  </select>
  <p className="text-xs text-gray-500 mt-1">
    Leave empty to let the routing engine select the best model automatically
  </p>
</div>
```

### 4.6 Create `frontend/src/pages/ModelComparison.tsx`

```typescript
/**
 * Model Comparison page.
 */
import React, { useState, useEffect } from 'react';
import { modelsApi } from '../api/modelsApi';
import type { ModelProfile } from '../types/routing';

export const ModelComparison: React.FC = () => {
  const [models, setModels] = useState<ModelProfile[]>([]);
  const [comparison, setComparison] = useState<any>(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    loadData();
  }, []);

  const loadData = async () => {
    setLoading(true);
    try {
      const comparisonData = await modelsApi.compareModels();
      setModels(comparisonData.models);
      setComparison(comparisonData);
    } catch (error) {
      console.error('Failed to load models:', error);
    } finally {
      setLoading(false);
    }
  };

  if (loading) {
    return <div className="p-8">Loading models...</div>;
  }

  return (
    <div className="p-8">
      <h1 className="text-3xl font-bold mb-6">Model Comparison</h1>

      {/* Highlights */}
      <div className="grid grid-cols-3 gap-6 mb-8">
        <div className="bg-green-50 border border-green-200 rounded-lg p-4">
          <h3 className="text-sm font-medium text-green-800 mb-2">💰 Most Cost-Effective</h3>
          <p className="text-2xl font-bold text-green-900">{comparison?.cheapest || 'N/A'}</p>
        </div>
        <div className="bg-blue-50 border border-blue-200 rounded-lg p-4">
          <h3 className="text-sm font-medium text-blue-800 mb-2">⚡ Fastest</h3>
          <p className="text-2xl font-bold text-blue-900">{comparison?.fastest || 'N/A'}</p>
        </div>
        <div className="bg-purple-50 border border-purple-200 rounded-lg p-4">
          <h3 className="text-sm font-medium text-purple-800 mb-2">🏆 Best for Code</h3>
          <p className="text-2xl font-bold text-purple-900">
            {comparison?.most_capable?.code_generation || 'N/A'}
          </p>
        </div>
      </div>

      {/* Models Table */}
      <div className="bg-white rounded-lg shadow-md overflow-hidden">
        <table className="w-full">
          <thead className="bg-gray-100">
            <tr>
              <th className="px-4 py-3 text-left text-sm font-medium">Model</th>
              <th className="px-4 py-3 text-left text-sm font-medium">Provider</th>
              <th className="px-4 py-3 text-center text-sm font-medium">Context</th>
              <th className="px-4 py-3 text-center text-sm font-medium">Latency</th>
              <th className="px-4 py-3 text-center text-sm font-medium">Cost (1K tokens)</th>
              <th className="px-4 py-3 text-center text-sm font-medium">Analysis</th>
              <th className="px-4 py-3 text-center text-sm font-medium">Code Gen</th>
            </tr>
          </thead>
          <tbody>
            {models.map((model) => (
              <tr key={model.id} className="border-b border-gray-200 hover:bg-gray-50">
                <td className="px-4 py-3">
                  <div className="font-medium">{model.display_name}</div>
                  <div className="text-xs text-gray-500">{model.model_name}</div>
                </td>
                <td className="px-4 py-3 capitalize">{model.provider}</td>
                <td className="px-4 py-3 text-center text-sm">
                  {(model.max_context_tokens / 1000).toFixed(0)}K
                </td>
                <td className="px-4 py-3 text-center text-sm">
                  {model.avg_latency_ms ? `${model.avg_latency_ms}ms` : 'N/A'}
                </td>
                <td className="px-4 py-3 text-center text-sm">
                  ${((model.cost_per_input_token + model.cost_per_output_token) * 500).toFixed(4)}
                </td>
                <td className="px-4 py-3 text-center">
                  <CapabilityBadge score={model.capabilities.analysis} />
                </td>
                <td className="px-4 py-3 text-center">
                  <CapabilityBadge score={model.capabilities.code_generation} />
                </td>
              </tr>
            ))}
          </tbody>
        </table>
      </div>
    </div>
  );
};

const CapabilityBadge: React.FC<{ score: number }> = ({ score }) => {
  const percentage = (score * 100).toFixed(0);
  const color = score >= 0.9 ? 'green' : score >= 0.8 ? 'blue' : 'gray';
  
  return (
    <span className={`px-2 py-1 bg-${color}-100 text-${color}-700 rounded text-xs font-medium`}>
      {percentage}%
    </span>
  );
};
```
