@pytest.mark.asyncio
@patch("backend.app.services.llm.openai_provider.OpenAIProvider.generate")
@patch("backend.app.services.llm.anthropic_provider.AnthropicProvider.generate")
async def test_e2e_routing_in_execution(
    mock_anthropic: AsyncMock,
    mock_openai: AsyncMock,
    db_session: AsyncSession,
    mock_llm_response
):
    """End-to-end: Create workflow → Seed models/rules → Execute → Verify routing."""
    # Seed models and rules
    await seed_model_profiles(db_session)
    await seed_default_routing_rules(db_session)
    
    # Create workflow with multiple stage types
    workflow = Workflow(name="E2E Routing Test", objective="Test routing in execution")
    db_session.add(workflow)
    await db_session.flush()
    
    stage1 = Stage(
        workflow_id=workflow.id, name="Analysis",
        instruction="Analyze requirements", stage_order=0, stage_type="analysis"
    )
    stage2 = Stage(
        workflow_id=workflow.id, name="Code Generation",
        instruction="Generate code", stage_order=1, stage_type="code_generation"
    )
    db_session.add_all([stage1, stage2])
    await db_session.commit()
    
    # Mock LLM responses
    mock_openai.return_value = mock_llm_response("Analysis result")
    mock_anthropic.return_value = mock_llm_response("Code result")
    
    # Execute
    engine = ExecutionEngine(db_session)
    execution_id = await engine.execute_workflow(workflow.id)
    await db_session.commit()
    
    # Verify routing decisions logged
    from sqlalchemy import select
    from backend.app.models.routing_rule import RoutingDecision
    
    result = await db_session.execute(
        select(RoutingDecision)
        .where(RoutingDecision.execution_id == execution_id)
        .order_by(RoutingDecision.created_at)
    )
    decisions = result.scalars().all()
    
    assert len(decisions) == 2
    
    # Stage 1 (analysis) - should use fast/balanced model
    assert decisions[0].stage_type == "analysis"
    assert decisions[0].selected_model_name is not None
    
    # Stage 2 (code_generation) - should use quality model
    assert decisions[1].stage_type == "code_generation"
    assert decisions[1].selected_model_name is not None
    
    print(f"Stage 1: {decisions[0].selected_model_name} - {decisions[0].selection_reason}")
    print(f"Stage 2: {decisions[1].selected_model_name} - {decisions[1].selection_reason}")

@pytest.mark.asyncio
async def test_routing_decision_logging(db_session: AsyncSession):
    """Test that all routing decisions are logged."""
    await seed_model_profiles(db_session)
    await seed_default_routing_rules(db_session)
    
    workflow = Workflow(name="Logging Test")
    db_session.add(workflow)

    stages = [

              instruction=f"Task {i}", stage_order=i, stage_type="analysis")

    # Verify all decisions logged

    from backend.app.models.routing_rule import RoutingDecision
    
    result = await db_session.execute(
        select(RoutingDecision).where(RoutingDecision.execution_id == execution_id)
    )
    decisions = result.scalars().all()
    
    assert len(decisions) == 3
    assert all(d.selected_model_name is not None for d in decisions)
    assert all(d.selection_reason is not None for d in decisions)
