@pytest.mark.asyncio
async def test_embedding_similarity_for_semantically_similar_text():
    """Test that semantically similar texts produce similar embeddings."""
    service = EmbeddingService(provider="openai")
    
    text1 = "Design a REST API for user authentication"
    text2 = "Create a RESTful API for authenticating users"
    text3 = "How to make pasta carbonara"
    
    emb1 = await service.generate_embedding(text1)
    emb2 = await service.generate_embedding(text2)
    emb3 = await service.generate_embedding(text3)
    
    # Calculate cosine similarity manually (simplified)
    def cosine_similarity(a, b):
        import math
        dot = sum(x * y for x, y in zip(a, b))
        mag_a = math.sqrt(sum(x * x for x in a))
        mag_b = math.sqrt(sum(y * y for y in b))
        return dot / (mag_a * mag_b)
    
    sim_12 = cosine_similarity(emb1, emb2)
    sim_13 = cosine_similarity(emb1, emb3)
    
    # Similar texts should have higher similarity
    assert sim_12 > 0.90  # Very similar
    assert sim_13 < 0.70  # Not similar

@pytest.mark.asyncio
async def test_dependency_hash_invalidates_cache(db_session: AsyncSession, sample_workflow: Workflow):
    """Test that cache is invalidated when dependencies change."""
    cache_service = SemanticCacheService(db_session)
    validator = CacheValidator(db_session)
    
    stage1 = sample_workflow.stages[0]
    stage2 = sample_workflow.stages[1]
    execution_id = uuid.uuid4()
    
    # Add stage 1 context
    from backend.app.services.context_manager import ContextManager
    ctx_mgr = ContextManager(db_session)
    await ctx_mgr.add_context(
        sample_workflow.id, execution_id, stage1.id,
        content="Stage 1 output: Requirements A, B, C",
        context_type="stage_output"
    )
    await db_session.commit()
    
    # Cache stage 2 with dependency hash
    hash1 = await validator.compute_dependency_hash(sample_workflow.id, execution_id, stage2)
    entry = await cache_service.store(
        input_text="Design based on requirements",
        result="Design result",
        stage_type=stage2.stage_type,
        dependency_hash=hash1
    )
    await db_session.commit()
    
    # Change stage 1 output
    await ctx_mgr.add_context(
        sample_workflow.id, execution_id, stage1.id,
        content="Stage 1 output: Requirements X, Y, Z",  # Changed
        context_type="stage_output"
    )
    await db_session.commit()
    
    # Compute new hash
    hash2 = await validator.compute_dependency_hash(sample_workflow.id, execution_id, stage2)
    
    # Hashes should differ
    assert hash1 != hash2
    
    # Cache should be invalid
    is_valid = await validator.is_cache_valid(entry, hash2)
    assert is_valid is False

@pytest.mark.asyncio
@patch("backend.app.services.llm.openai_provider.OpenAIProvider.generate")
async def test_execution_uses_cache_second_run(
    mock_generate: AsyncMock,
    db_session: AsyncSession,
    sample_workflow: Workflow,
    mock_llm_response
):
    """Test that second execution uses cache, skipping LLM calls."""
    # First execution
    mock_generate.side_effect = [
        mock_llm_response("Stage 1 result"),
        mock_llm_response("Stage 2 result"),
        mock_llm_response("Stage 3 result")
    ]
    
    engine = ExecutionEngine(db_session)
    exec1 = await engine.execute_workflow(sample_workflow.id)
    await db_session.commit()
    
    llm_calls_first = mock_generate.call_count
    assert llm_calls_first == 3
    
    # Reset mock
    mock_generate.reset_mock()
    mock_generate.side_effect = []  # Should not be called
    
    # Second execution (should hit cache)
    exec2 = await engine.execute_workflow(sample_workflow.id)
    await db_session.commit()
    
    llm_calls_second = mock_generate.call_count
    assert llm_calls_second == 0  # No LLM calls (all cache hits)
    
    # Verify execution records show cache hits
    from sqlalchemy import select
    from backend.app.models.execution import ExecutionRecord
    
    result = await db_session.execute(
        select(ExecutionRecord).where(ExecutionRecord.execution_id == exec2)
    )
    records = result.scalars().all()
    
    # All records should have provider="cache"
    assert all(r.provider == "cache" for r in records)

    # Calculate hit rate
    from sqlalchemy import select, func
    from backend.app.models.cache import CacheEntry
    
    result = await db_session.execute(
        select(func.count(CacheEntry.id), func.sum(CacheEntry.hit_count))
        .where(CacheEntry.is_valid == True)
    )
    total_entries, total_hits = result.first()
    
    total_lookups = total_entries + total_hits
    hit_rate = total_hits / total_lookups if total_lookups > 0 else 0
    
    assert hit_rate > 0.5  # At least 50% hit rate

async def test_e2e_workflow_with_caching(
    mock_generate: AsyncMock,
    db_session: AsyncSession,
    mock_llm_response
):
    """
    End-to-end test: Create workflow → Execute → Cache results → Re-execute → Verify cache hits.
    """
    # Create workflow
    workflow = Workflow(
        name="E2E Cache Test",
        objective="Test full caching pipeline"
    )
    db_session.add(workflow)
    await db_session.flush()
    
    # Add 2 stages
    stage1 = Stage(
        workflow_id=workflow.id, name="Analyze", instruction="Analyze requirements",
        stage_order=0, stage_type="analysis"
    )
    stage2 = Stage(
        workflow_id=workflow.id, name="Design", instruction="Design architecture",
        stage_order=1, stage_type="design"
    )
    db_session.add_all([stage1, stage2])
    await db_session.commit()
    
    # Mock LLM responses
    mock_generate.side_effect = [
        mock_llm_response("Requirements: Auth, CRUD"),
        mock_llm_response("Architecture: REST + DB")
    ]
    
    # First execution
    engine = ExecutionEngine(db_session)
    exec1 = await engine.execute_workflow(workflow.id)
    await db_session.commit()
    
    assert mock_generate.call_count == 2
    
    # Verify cache entries created
    from sqlalchemy import select
    from backend.app.models.cache import CacheEntry
    
    result = await db_session.execute(
        select(func.count(CacheEntry.id)).where(CacheEntry.workflow_id == workflow.id)
    )
    cache_count = result.scalar()
    assert cache_count == 2  # Both stages cached
    
    # Reset mock
    mock_generate.reset_mock()
    
    # Second execution (should hit cache)
    exec2 = await engine.execute_workflow(workflow.id)
    await db_session.commit()
    
    assert mock_generate.call_count == 0  # No LLM calls
    
    # Verify results identical
    from backend.app.models.execution import ExecutionRecord
    
    result1 = await db_session.execute(
        select(ExecutionRecord).where(ExecutionRecord.execution_id == exec1)
    )
    records1 = result1.scalars().all()
    
    result2 = await db_session.execute(
        select(ExecutionRecord).where(ExecutionRecord.execution_id == exec2)
    )
    records2 = result2.scalars().all()
    
    assert len(records1) == len(records2)
    assert records1[0].result == records2[0].result
    assert records2[0].provider == "cache"
