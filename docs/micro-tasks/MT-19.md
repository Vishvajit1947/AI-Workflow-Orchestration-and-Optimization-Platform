# MT-19 — Semantic Cache Model & Service

## 1. Context
With embeddings working (MT-18), we now create the semantic cache storage layer. The `cache_entries` table stores stage inputs, their embeddings, and the LLM results. The cache service provides lookup (via cosine similarity search) and store operations.

## 2. Prerequisites
| What | Why |
|------|-----|
| **MT-18 complete** | Embedding service available |
| **MT-03 complete** | Alembic migrations exist |
| **pgvector enabled** | Vector similarity search |

## 3. Goal
- `backend/app/models/cache.py` — CacheEntry ORM model with vector column.
- Alembic migration to create `cache_entries` table with vector index.
- `backend/app/services/cache/semantic_cache.py` — Cache lookup and store logic.
- Cosine similarity search using pgvector.
- Configurable similarity threshold (default 0.92).
- Cache hit tracking (increment hit_count).
- Pydantic schemas for cache operations.

## 4. Implementation

```python
"""
Cache Entry ORM model.
"""
import uuid
from datetime import datetime, timezone
from typing import List

from sqlalchemy import (
    Column, String, Text, Integer, DateTime, ForeignKey,
    Boolean, Numeric, Index, UUID
)
from pgvector.sqlalchemy import Vector

from backend.app.database import Base


class CacheEntry(Base):
    __tablename__ = "cache_entries"

    id = Column(UUID(as_uuid=True), primary_key=True, default=uuid.uuid4)
    workflow_id = Column(UUID(as_uuid=True), ForeignKey("workflows.id", ondelete="CASCADE"), nullable=True)
    stage_id = Column(UUID(as_uuid=True), ForeignKey("stages.id", ondelete="CASCADE"), nullable=True)
    stage_type = Column(String(100), nullable=True)  # analysis, design, generation, etc.
    
    input_text = Column(Text, nullable=False)
    input_embedding = Column(Vector(1536), nullable=False)  # Embedding dimension
    
    result = Column(Text, nullable=False)
    result_tokens = Column(Integer, nullable=True)
    model_used = Column(String(100), nullable=True)
    
    dependency_hash = Column(String(64), nullable=True)  # Hash of upstream outputs
    hit_count = Column(Integer, default=0)
    similarity_threshold = Column(Numeric(4, 3), default=0.92)
    
    is_valid = Column(Boolean, default=True)
    
    created_at = Column(DateTime(timezone=True), default=lambda: datetime.now(timezone.utc))
    expires_at = Column(DateTime(timezone=True), nullable=True)
    
    # Indexes for fast lookup
    __table_args__ = (
        Index("idx_cache_entries_stage_type", "stage_type"),
        Index("idx_cache_entries_is_valid", "is_valid"),
        Index("idx_cache_entries_created_at", "created_at"),
        # Vector index for similarity search (ivfflat algorithm)
        Index(
            "idx_cache_entries_embedding",
            "input_embedding",
            postgresql_using="ivfflat",
            postgresql_with={"lists": 100},
            postgresql_ops={"input_embedding": "vector_cosine_ops"}
        ),
    )
    
    def __repr__(self):
        return f"<CacheEntry(id={self.id}, stage_type={self.stage_type}, hits={self.hit_count})>"
```

    stage_id: Optional[uuid.UUID] = None
    stage_type: Optional[str] = None
    input_text: str
    input_embedding: List[float]

    result_tokens: Optional[int] = None
    model_used: Optional[str] = None
    dependency_hash: Optional[str] = None
    similarity_threshold: Decimal = Field(default=Decimal("0.92"))

    """Schema for cache entry response."""
    id: uuid.UUID
    stage_type: Optional[str]
    input_text: str
    result: str
    result_tokens: Optional[int]
    model_used: Optional[str]

    created_at: datetime
    
    class Config:
        from_attributes = True

    similarity_score: Optional[float] = None
    tokens_saved: Optional[int] = None

class CacheStats(BaseModel):

    total_entries: int

    hit_rate: float
    total_tokens_saved: int

    avg_similarity_score: float

### 4.4 Create `backend/app/services/cache/semantic_cache.py`

```python
"""
Semantic Cache Service.
Handles cache lookup via similarity search and cache storage.
"""
import hashlib
import uuid
from datetime import datetime, timezone, timedelta
from decimal import Decimal
from typing import Optional, List

from sqlalchemy import select, update, func, text
from sqlalchemy.ext.asyncio import AsyncSession

from backend.app.config import settings
from backend.app.models.cache import CacheEntry
from backend.app.schemas.cache import CacheEntryCreate, CacheEntryResponse, CacheHitResponse
from backend.app.services.embedding_service import get_embedding_service


class SemanticCacheService:
    """Service for semantic caching with vector similarity search."""
    
    def __init__(self, db: AsyncSession):
        self.db = db
        self.embedding_service = get_embedding_service()
        self.similarity_threshold = float(settings.CACHE_SIMILARITY_THRESHOLD)
    
    async def lookup(
        self,
        input_text: str,
        stage_type: Optional[str] = None,
        dependency_hash: Optional[str] = None,
        similarity_threshold: Optional[float] = None
    ) -> CacheHitResponse:
        """
        Look up cache entry via semantic similarity search.
        
        Args:
            input_text: The input text to search for
            stage_type: Optional filter by stage type
            dependency_hash: Optional dependency hash for validation
            similarity_threshold: Override default threshold
            
        Returns:
            CacheHitResponse with cache_hit=True if found
        """
        threshold = similarity_threshold or self.similarity_threshold
        
        # Generate embedding for input
        embedding = await self.embedding_service.generate_embedding(input_text)
        
        # Build similarity search query
        query = select(
            CacheEntry,
            # Calculate cosine similarity (1 - cosine_distance)
            (1 - CacheEntry.input_embedding.cosine_distance(embedding)).label("similarity")
        ).where(
            CacheEntry.is_valid == True,
            # Similarity threshold filter
            (1 - CacheEntry.input_embedding.cosine_distance(embedding)) >= threshold
        )
        
        # Optional filters
        if stage_type:
            query = query.where(CacheEntry.stage_type == stage_type)
        
        if dependency_hash:
            query = query.where(CacheEntry.dependency_hash == dependency_hash)
        
        # Check expiry
        query = query.where(
            (CacheEntry.expires_at.is_(None)) | (CacheEntry.expires_at > datetime.now(timezone.utc))
        )
        
        # Order by similarity descending, limit to best match
        query = query.order_by(text("similarity DESC")).limit(1)
        
        result = await self.db.execute(query)
        row = result.first()
        
        if not row:
            return CacheHitResponse(cache_hit=False)
        
        cache_entry, similarity_score = row
        
        # Increment hit count
        await self._increment_hit_count(cache_entry.id)
        
        # Calculate tokens and cost saved (estimate)
        tokens_saved = cache_entry.result_tokens or 0
        cost_saved = self._estimate_cost_saved(tokens_saved)
        
        entry_response = CacheEntryResponse.model_validate(cache_entry)
        entry_response.similarity_score = float(similarity_score)
        
        return CacheHitResponse(
            cache_hit=True,
            entry=entry_response,
            similarity_score=float(similarity_score),
            tokens_saved=tokens_saved,
            cost_saved=cost_saved
        )
    
    async def store(
        self,
        input_text: str,
        result: str,
        stage_type: Optional[str] = None,
        workflow_id: Optional[uuid.UUID] = None,
        stage_id: Optional[uuid.UUID] = None,
        result_tokens: Optional[int] = None,
        model_used: Optional[str] = None,
        dependency_hash: Optional[str] = None,
        ttl_seconds: Optional[int] = None
    ) -> CacheEntry:
        """
        Store a new cache entry.
        
        Args:
            input_text: The input text
            result: The LLM result to cache
            stage_type: Type of stage (analysis, design, etc.)
            workflow_id: Associated workflow ID
            stage_id: Associated stage ID
            result_tokens: Number of tokens in result
            model_used: Model that generated the result
            dependency_hash: Hash of upstream dependencies
            ttl_seconds: Time-to-live in seconds (None = no expiry)
            
        Returns:
            Created CacheEntry
        """
        # Generate embedding
        embedding = await self.embedding_service.generate_embedding(input_text)
        
        # Calculate expiry
        expires_at = None
        if ttl_seconds:
            expires_at = datetime.now(timezone.utc) + timedelta(seconds=ttl_seconds)
        
        # Create cache entry
        cache_entry = CacheEntry(
            workflow_id=workflow_id,
            stage_id=stage_id,
            stage_type=stage_type,
            input_text=input_text,
            input_embedding=embedding,
            result=result,
            result_tokens=result_tokens,
            model_used=model_used,
            dependency_hash=dependency_hash,
            similarity_threshold=Decimal(str(self.similarity_threshold)),
            expires_at=expires_at
        )
        
        self.db.add(cache_entry)
        await self.db.flush()
        
        return cache_entry
    
    async def invalidate(
        self,
        workflow_id: Optional[uuid.UUID] = None,
        stage_id: Optional[uuid.UUID] = None,
        stage_type: Optional[str] = None
    ) -> int:
        """
        Invalidate cache entries by marking them as invalid.
        
        Args:
            workflow_id: Invalidate all entries for a workflow
            stage_id: Invalidate all entries for a stage
            stage_type: Invalidate all entries of a stage type
            
        Returns:
            Number of entries invalidated
        """
        query = update(CacheEntry).where(CacheEntry.is_valid == True).values(is_valid=False)
        
        if workflow_id:
            query = query.where(CacheEntry.workflow_id == workflow_id)
        if stage_id:
            query = query.where(CacheEntry.stage_id == stage_id)
        if stage_type:
            query = query.where(CacheEntry.stage_type == stage_type)
        
        result = await self.db.execute(query)
        await self.db.flush()
        
        return result.rowcount
    
    async def clear_expired(self) -> int:
        """
        Remove expired cache entries.
        
        Returns:
            Number of entries removed
        """
        query = update(CacheEntry).where(
            CacheEntry.expires_at.isnot(None),
            CacheEntry.expires_at < datetime.now(timezone.utc)
        ).values(is_valid=False)
        
        result = await self.db.execute(query)
        await self.db.flush()
        
        return result.rowcount
    
    async def _increment_hit_count(self, entry_id: uuid.UUID):
        """Increment the hit count for a cache entry."""
        await self.db.execute(
            update(CacheEntry)
            .where(CacheEntry.id == entry_id)
            .values(hit_count=CacheEntry.hit_count + 1)
        )
        await self.db.flush()
    
    @staticmethod
    def _estimate_cost_saved(tokens: int, model: str = "gpt-4o-mini") -> Decimal:
        """Estimate cost saved by cache hit (simplified)."""
        # Simplified cost estimation (output tokens)
        cost_per_token = Decimal("0.000001")  # $0.001 per 1K tokens = $0.000001 per token
        return Decimal(tokens) * cost_per_token
    
    @staticmethod
    def compute_dependency_hash(dependency_outputs: List[str]) -> str:
        """
        Compute hash of upstream stage outputs.
        Used for cache invalidation when dependencies change.
        """
        combined = "||".join(dependency_outputs)
        return hashlib.sha256(combined.encode()).hexdigest()
```

"""

import pytest
from sqlalchemy.ext.asyncio import AsyncSession

from backend.app.services.cache.semantic_cache import SemanticCacheService


@pytest.mark.asyncio
async def test_cache_store_and_lookup(db_session: AsyncSession):
    """Test storing and retrieving cache entries."""
    cache_service = SemanticCacheService(db_session)
    
    # Store entry
    entry = await cache_service.store(
        input_text="What are the key features of a todo app?",
        result="Key features: create, read, update, delete tasks",
        stage_type="analysis",
        result_tokens=50,
        model_used="gpt-4o-mini"
    )
    await db_session.commit()
    
    assert entry.id is not None
    
    # Lookup with exact text
    result = await cache_service.lookup(
        input_text="What are the key features of a todo app?",
        stage_type="analysis"
    )
    
    assert result.cache_hit is True
    assert result.entry.result == "Key features: create, read, update, delete tasks"
    assert result.similarity_score >= 0.99  # Very high similarity

    """Test that hit count increments on cache hits."""
