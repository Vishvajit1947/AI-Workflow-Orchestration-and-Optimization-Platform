# MT-37 — Analytics API Endpoints

## 1. Context
With analytics aggregation service complete (MT-36), this MT exposes analytics data through REST API endpoints for frontend consumption.

## 2. Prerequisites
| What | Why |
|------|-----|
| **MT-36 complete** | Analytics aggregator available |
| **MT-15 complete** | Execution records exist |

## 3. Goal
- `backend/app/routers/analytics.py` — Analytics REST endpoints
- `/api/analytics/overview` — Key platform metrics
- `/api/analytics/workflows/{id}` — Per-workflow metrics
- `/api/analytics/cache` — Cache performance
- `/api/analytics/models` — Model utilization

"""
Analytics API Router.
Exposes analytics endpoints for frontend consumption.
"""
from datetime import datetime, timedelta
from typing import Optional, Literal
from uuid import UUID

from fastapi import APIRouter, Depends, Query, HTTPException
from fastapi.responses import StreamingResponse
from sqlalchemy.ext.asyncio import AsyncSession

from backend.app.database import get_db
from backend.app.services.analytics.aggregator import AnalyticsAggregator
from backend.app.services.analytics.export import AnalyticsExporter
from backend.app.schemas.analytics import (
    WorkflowMetrics,
    ModelUtilization,
    CacheMetrics,
    CostTrendPoint,
    TokenTrendPoint,
    RoutingEffectiveness
)


router = APIRouter(prefix="/api/analytics", tags=["analytics"])


@router.get("/overview", response_model=WorkflowMetrics)
async def get_overview_metrics(
    start_date: Optional[datetime] = Query(None),
    end_date: Optional[datetime] = Query(None),
    db: AsyncSession = Depends(get_db)
):
    """
    Get overview metrics for the entire platform.
    
    Args:
        start_date: Optional start date filter
        end_date: Optional end date filter
        
    Returns:
        WorkflowMetrics with aggregated platform statistics
    """
    aggregator = AnalyticsAggregator(db)
    metrics = await aggregator.get_workflow_metrics(
        workflow_id=None,
        start_date=start_date,
        end_date=end_date
    )
    return WorkflowMetrics(**metrics)


@router.get("/workflows/{workflow_id}", response_model=WorkflowMetrics)

@router.get("/timeline/{execution_id}")
async def get_execution_timeline(
    execution_id: UUID,
    db: AsyncSession = Depends(get_db)
):
    """
    Get execution timeline for Gantt chart visualization.
    
    Args:
        execution_id: Execution UUID
        
    Returns:
        Timeline data with stage start/end times
    """
    from sqlalchemy import select
    from backend.app.models.execution import ExecutionRecord
    from backend.app.models.stage import Stage
    
    # Get all execution records for this execution
    query = select(ExecutionRecord).where(
        ExecutionRecord.execution_id == execution_id
    ).order_by(ExecutionRecord.started_at)
    
    result = await db.execute(query)
    records = result.scalars().all()
    
    if not records:
        raise HTTPException(status_code=404, detail="Execution not found")
    
    # Load stages for names
    stage_ids = [r.stage_id for r in records if r.stage_id]
    stage_query = select(Stage).where(Stage.id.in_(stage_ids))
    stage_result = await db.execute(stage_query)
    stages = {s.id: s for s in stage_result.scalars().all()}
    
    timeline = []
    for record in records:
        stage = stages.get(record.stage_id)
        timeline.append({
            "stage_id": str(record.stage_id),
            "stage_name": stage.name if stage else "Unknown",
            "started_at": record.started_at.isoformat() if record.started_at else None,
            "completed_at": record.completed_at.isoformat() if record.completed_at else None,
            "duration_ms": record.latency_ms,
            "status": record.status,
            "model_used": record.model_used,
            "provider": record.provider
        })
    
    return {
        "execution_id": str(execution_id),
        "timeline": timeline
    }

async def get_routing_effectiveness(db: AsyncSession = Depends(get_db)):
    """

    """
    aggregator = AnalyticsAggregator(db)
    metrics = await aggregator.get_routing_effectiveness()
    return RoutingEffectiveness(**metrics)


@router.get("/export")
async def export_analytics(
    format: Literal["csv", "json"] = Query(default="json"),
    start_date: Optional[datetime] = Query(None),
    end_date: Optional[datetime] = Query(None),
    db: AsyncSession = Depends(get_db)
):
    """
    Export analytics data as CSV or JSON.
    
    Args:
        format: Export format (csv or json)
        start_date: Optional start date filter
        end_date: Optional end date filter
        
    Returns:
        StreamingResponse with file content
    """
    exporter = AnalyticsExporter(db)
    
    if format == "csv":
        content = await exporter.export_csv(start_date, end_date)
        media_type = "text/csv"
        filename = f"analytics_{datetime.now().strftime('%Y%m%d')}.csv"
    else:
        content = await exporter.export_json(start_date, end_date)
        media_type = "application/json"
        filename = f"analytics_{datetime.now().strftime('%Y%m%d')}.json"
    
    return StreamingResponse(
        iter([content]),
        media_type=media_type,
        headers={
            "Content-Disposition": f"attachment; filename={filename}"
        }
    )

### 4.2 Create `backend/app/services/analytics/export.py`

```python
"""
Analytics Export Service.
Exports analytics data in various formats.
"""
import csv
import json
from io import StringIO
from datetime import datetime
from typing import Optional

from sqlalchemy.ext.asyncio import AsyncSession

from backend.app.services.analytics.aggregator import AnalyticsAggregator


class AnalyticsExporter:
    """Service for exporting analytics data."""
    
    def __init__(self, db: AsyncSession):
        self.db = db
        self.aggregator = AnalyticsAggregator(db)
    
    async def export_csv(
        self,
        start_date: Optional[datetime] = None,
        end_date: Optional[datetime] = None
    ) -> str:
        """
        Export analytics as CSV.
        
        Args:
            start_date: Optional start date filter
            end_date: Optional end date filter
            
        Returns:
            CSV string
        """
        output = StringIO()
        writer = csv.writer(output)
        
        # Workflow metrics
        writer.writerow(["=== Workflow Metrics ==="])
        metrics = await self.aggregator.get_workflow_metrics(
            start_date=start_date,
            end_date=end_date
        )
        writer.writerow(["Metric", "Value"])
        for key, value in metrics.items():
            writer.writerow([key, value])
        
        writer.writerow([])
        
        # Model utilization
        writer.writerow(["=== Model Utilization ==="])
        models = await self.aggregator.get_model_utilization(
            start_date=start_date,
            end_date=end_date
        )
        if models:
            writer.writerow(models[0].keys())
            for model in models:
                writer.writerow(model.values())
        
        writer.writerow([])
        
        # Cache metrics
        writer.writerow(["=== Cache Metrics ==="])
        cache = await self.aggregator.get_cache_metrics()
        writer.writerow(["Metric", "Value"])
        for key, value in cache.items():
            writer.writerow([key, value])
        
        return output.getvalue()
    
    async def export_json(
        self,
        start_date: Optional[datetime] = None,
        end_date: Optional[datetime] = None
    ) -> str:
        """
        Export analytics as JSON.
        
        Args:
            start_date: Optional start date filter
            end_date: Optional end date filter
            
        Returns:
            JSON string
        """
        data = {
            "exported_at": datetime.now().isoformat(),
            "filters": {
                "start_date": start_date.isoformat() if start_date else None,
                "end_date": end_date.isoformat() if end_date else None
            },
            "workflow_metrics": await self.aggregator.get_workflow_metrics(
                start_date=start_date,
                end_date=end_date
            ),
            "model_utilization": await self.aggregator.get_model_utilization(
                start_date=start_date,
                end_date=end_date
            ),
            "cache_metrics": await self.aggregator.get_cache_metrics(),
            "cost_trend": await self.aggregator.get_cost_trend(days=30),
            "token_trend": await self.aggregator.get_token_usage_trend(days=30)
        }
        
        return json.dumps(data, indent=2)
```
