# MT-20 — Workflow-Aware Cache Validation

## 1. Context
Basic semantic caching works (MT-19), but we need **workflow-aware validation**: if upstream stage outputs change, downstream cache entries become stale. This MT implements dependency-hash tracking and automatic cache invalidation when dependencies change.

## 2. Prerequisites
| What | Why |
|------|-----|
| **MT-19 complete** | Semantic cache service exists |
| **MT-14 complete** | Execution engine with context flow |
| **MT-11 complete** | Context manager tracks stage outputs |

## 3. Goal
- Dependency hash computation from upstream stage outputs.
- Cache validation logic that checks dependency hashes.
- Automatic cache invalidation when upstream stages are re-executed.
- TTL (time-to-live) expiry for cache entries.
- Cache warming strategy for common workflows.
- Enhanced cache service with workflow-aware lookup.

## 4. Implementation

### 4.1 Create `backend/app/services/cache/cache_validator.py`

```python
"""
Cache Validator.
Workflow-aware cache validation using dependency hashes and TTL.
"""
import hashlib
import uuid
from datetime import datetime, timezone
from typing import List, Optional

from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession

from backend.app.models.cache import CacheEntry
from backend.app.models.stage import Stage
from backend.app.services.context_manager import ContextManager


class CacheValidator:
    """Validates cache entries based on workflow dependencies."""
    
    def __init__(self, db: AsyncSession):
        self.db = db
        self.context_mgr = ContextManager(db)
    
    async def compute_dependency_hash(
        self,
        workflow_id: uuid.UUID,
        execution_id: uuid.UUID,
        current_stage: Stage
    ) -> str:
        """
        Compute dependency hash from upstream stage outputs.
        
        Args:
            workflow_id: Workflow ID
            execution_id: Current execution ID
            current_stage: Current stage (to get dependencies)
            
        Returns:
            SHA-256 hash of all dependency outputs
        """
        # Get all upstream dependencies for this stage
        dependency_outputs = []
        
        # Get contexts from stages that this stage depends on
        contexts = await self.context_mgr.get_relevant_context(
            workflow_id=workflow_id,
            execution_id=execution_id,
            current_stage_order=current_stage.stage_order,
            max_tokens=None  # Get all context
        )
        
        # Extract content in deterministic order
        for context in sorted(contexts, key=lambda c: c.created_at):
            if context.context_type == "stage_output":
                dependency_outputs.append(context.content)
        
        # Compute hash
        if not dependency_outputs:
            # No dependencies = empty hash
            return hashlib.sha256(b"").hexdigest()
        
        combined = "||".join(dependency_outputs)
        return hashlib.sha256(combined.encode()).hexdigest()
    
    async def is_cache_valid(
        self,
        cache_entry: CacheEntry,
        current_dependency_hash: Optional[str] = None
    ) -> bool:
        """
        Check if a cache entry is still valid.
        
        Args:
            cache_entry: The cache entry to validate
            current_dependency_hash: Current hash of dependencies
            
        Returns:
            True if cache is valid, False if stale
        """
        # Check if explicitly marked invalid
        if not cache_entry.is_valid:
            return False
        
        # Check TTL expiry
        if cache_entry.expires_at:
            if datetime.now(timezone.utc) > cache_entry.expires_at:
                return False
        
        # Check dependency hash if provided
        if current_dependency_hash and cache_entry.dependency_hash:
            if current_dependency_hash != cache_entry.dependency_hash:
                return False
        
        return True
    
    async def invalidate_downstream_cache(
        self,
        workflow_id: uuid.UUID,
        changed_stage_id: uuid.UUID
    ) -> int:
        """
        Invalidate cache entries for all stages downstream of the changed stage.
        
        When a stage is re-executed, all dependent stages' cache becomes stale.
        
        Args:
            workflow_id: Workflow ID
            changed_stage_id: Stage that was re-executed
            
        Returns:
            Number of cache entries invalidated
        """
        # Get the changed stage
        result = await self.db.execute(
            select(Stage).where(Stage.id == changed_stage_id)
        )
        changed_stage = result.scalar_one_or_none()
        if not changed_stage:
            return 0
        
        # Get all stages in this workflow
        result = await self.db.execute(
            select(Stage).where(Stage.workflow_id == workflow_id)
        )
        all_stages = result.scalars().all()
        
        # Find downstream stages (stages that depend on the changed stage)
        downstream_stage_ids = await self._find_downstream_stages(
            changed_stage, all_stages
        )
        
        if not downstream_stage_ids:
            return 0
        
        # Invalidate cache for downstream stages
        from sqlalchemy import update
        query = update(CacheEntry).where(
            CacheEntry.workflow_id == workflow_id,
            CacheEntry.stage_id.in_(downstream_stage_ids),
            CacheEntry.is_valid == True
        ).values(is_valid=False)
        
        result = await self.db.execute(query)
        await self.db.flush()
        
        return result.rowcount
    
    async def _find_downstream_stages(
        self,
        changed_stage: Stage,
        all_stages: List[Stage]
    ) -> List[uuid.UUID]:
        """
        Find all stages downstream of the changed stage.
        
        A stage is downstream if:
        1. It has a higher stage_order (sequential dependency)
        2. It explicitly depends on the changed stage
        """
        downstream = []
        
        for stage in all_stages:
            # Sequential dependency: later stages depend on earlier ones
            if stage.stage_order > changed_stage.stage_order:
                downstream.append(stage.id)
            
            # Explicit dependency check (if stage_dependencies table is used)
            # TODO: Add explicit dependency checking when implementing parallel execution
        
        return downstream
    
    async def cleanup_expired_entries(self) -> int:
        """
        Clean up expired cache entries.
        Should be run periodically (e.g., daily cron job).
        
        Returns:
            Number of entries cleaned up
        """
        from sqlalchemy import update
        query = update(CacheEntry).where(
            CacheEntry.expires_at.isnot(None),
            CacheEntry.expires_at < datetime.now(timezone.utc),
            CacheEntry.is_valid == True
        ).values(is_valid=False)
        
        result = await self.db.execute(query)
        await self.db.flush()
        
        return result.rowcount
```

```python
# Add import
from backend.app.services.cache.cache_validator import CacheValidator

# Update SemanticCacheService class
class SemanticCacheService:
    def __init__(self, db: AsyncSession):
        self.db = db
        self.embedding_service = get_embedding_service()
        self.validator = CacheValidator(db)  # Add validator
        self.similarity_threshold = float(settings.CACHE_SIMILARITY_THRESHOLD)
    
    # Add workflow-aware lookup method
    async def lookup_with_validation(
        self,
        input_text: str,
        workflow_id: uuid.UUID,
        execution_id: uuid.UUID,
        current_stage: 'Stage',
        similarity_threshold: Optional[float] = None
    ) -> CacheHitResponse:
        """
        Workflow-aware cache lookup with dependency validation.
        
        Args:
            input_text: Input text to search
            workflow_id: Current workflow ID
            execution_id: Current execution ID
            current_stage: Current stage being executed
            similarity_threshold: Override default threshold
            
        Returns:
            CacheHitResponse with validation status
        """
        # Compute current dependency hash
        dependency_hash = await self.validator.compute_dependency_hash(
            workflow_id, execution_id, current_stage
        )
        
        # Perform similarity search
        result = await self.lookup(
            input_text=input_text,
            stage_type=current_stage.stage_type,
            dependency_hash=dependency_hash,
            similarity_threshold=similarity_threshold
        )
        
        # If cache hit, validate
        if result.cache_hit and result.entry:
            # Fetch full cache entry
            from sqlalchemy import select
            from backend.app.models.cache import CacheEntry
            
            db_result = await self.db.execute(
                select(CacheEntry).where(CacheEntry.id == result.entry.id)
            )
            cache_entry = db_result.scalar_one()
            
            # Validate
            is_valid = await self.validator.is_cache_valid(
                cache_entry, dependency_hash
            )
            
            if not is_valid:
                # Cache is stale, return miss
                return CacheHitResponse(cache_hit=False)
        
        return result
    
    # Update store method to include dependency hash
    async def store_with_dependencies(
        self,
        input_text: str,
        result: str,
        workflow_id: uuid.UUID,
        execution_id: uuid.UUID,
        current_stage: 'Stage',
        result_tokens: Optional[int] = None,
        model_used: Optional[str] = None,
        ttl_seconds: Optional[int] = None
    ) -> CacheEntry:
        """
        Store cache entry with dependency hash.
        
        Args:
            input_text: Input text
            result: LLM result
            workflow_id: Workflow ID
            execution_id: Execution ID
            current_stage: Current stage
            result_tokens: Token count
            model_used: Model name
            ttl_seconds: Time-to-live
            
        Returns:
            Created CacheEntry
        """
        # Compute dependency hash
        dependency_hash = await self.validator.compute_dependency_hash(
            workflow_id, execution_id, current_stage
        )
        
        # Store with dependency hash
        return await self.store(
            input_text=input_text,
            result=result,
            stage_type=current_stage.stage_type,
            workflow_id=workflow_id,
            stage_id=current_stage.id,
            result_tokens=result_tokens,
            model_used=model_used,
            dependency_hash=dependency_hash,
            ttl_seconds=ttl_seconds
        )
```

```python
Cache Validator Tests.

import pytest

@pytest.mark.asyncio
async def test_dependency_hash_computation(db_session: AsyncSession):
    """Test dependency hash computation."""
    # Create workflow with 2 stages
    workflow = Workflow(name="Test", objective="Test dependency hash")
    db_session.add(workflow)
    await db_session.flush()
    
    stage1 = Stage(
        workflow_id=workflow.id, name="Stage 1",
        instruction="First stage", stage_order=0, stage_type="analysis"
    )
    stage2 = Stage(
        workflow_id=workflow.id, name="Stage 2",
        instruction="Second stage", stage_order=1, stage_type="design"
    )
    db_session.add_all([stage1, stage2])
    await db_session.flush()
    
    execution_id = uuid.uuid4()
    
    # Add context from stage 1
    ctx_mgr = ContextManager(db_session)
    await ctx_mgr.add_context(
        workflow.id, execution_id, stage1.id,
        content="Stage 1 output", context_type="stage_output"
    )
    await db_session.commit()
    
    # Compute hash for stage 2
    validator = CacheValidator(db_session)
    hash1 = await validator.compute_dependency_hash(workflow.id, execution_id, stage2)
    
    assert hash1 is not None
    assert len(hash1) == 64  # SHA-256 hex digest
    
    # Add more context, hash should change
    await ctx_mgr.add_context(
        workflow.id, execution_id, stage1.id,
        content="Stage 1 updated output", context_type="stage_output"
    )
    await db_session.commit()
    
    hash2 = await validator.compute_dependency_hash(workflow.id, execution_id, stage2)
    assert hash2 != hash1  # Hash changed


@pytest.mark.asyncio
async def test_cache_invalidation_on_upstream_change(db_session: AsyncSession):
    """Test that cache is invalidated when upstream stage changes."""
    # Create workflow
    workflow = Workflow(name="Test", objective="Test cache invalidation")
    db_session.add(workflow)
    await db_session.flush()
    
    stage1 = Stage(
        workflow_id=workflow.id, name="Stage 1",
        instruction="Analysis", stage_order=0, stage_type="analysis"
    )
    stage2 = Stage(
        workflow_id=workflow.id, name="Stage 2",
        instruction="Design", stage_order=1, stage_type="design"
    )
    db_session.add_all([stage1, stage2])
    await db_session.flush()
    
    execution_id = uuid.uuid4()
    
    # Add stage 1 context
    ctx_mgr = ContextManager(db_session)
    await ctx_mgr.add_context(
        workflow.id, execution_id, stage1.id,
        content="Initial requirements", context_type="stage_output"
    )
    await db_session.commit()
    
    # Cache stage 2 result
    cache_service = SemanticCacheService(db_session)
    entry = await cache_service.store_with_dependencies(
        input_text="Design based on requirements",
        result="Initial design",
        workflow_id=workflow.id,
        execution_id=execution_id,
        current_stage=stage2,
        model_used="gpt-4o-mini"
    )
    await db_session.commit()

    validator = CacheValidator(db_session)
    invalidated = await validator.invalidate_downstream_cache(workflow.id, stage1.id)
    await db_session.commit()
    
    assert invalidated == 1  # Stage 2 cache invalidated
    
    await db_session.refresh(entry)
    assert entry.is_valid is False
```
