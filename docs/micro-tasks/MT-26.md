# MT-26 — Router Integration into Execution

## 1. Context
With the routing engine complete (MT-25), we now integrate it into the execution flow. Instead of using default models, the execution engine will consult the router to select the optimal model for each stage based on routing rules and stage requirements.

## 2. Prerequisites
| What | Why |
|------|-----|
| **MT-25 complete** | Routing engine available |
| **MT-14 complete** | Execution engine exists |
| **MT-21 complete** | Cache integration complete |

## 3. Goal
- Update execution engine to use routing engine for model selection.
- Pass stage and execution context to router.
- Support user overrides via execution request.
- Log routing decisions alongside execution records.
- Update execution API to accept routing preferences.
- Track model used per stage in execution records.

### 4.1 Update `backend/app/services/execution/execution_engine.py`

Integrate router into execution flow:

```python
# Add import
from backend.app.services.router.routing_engine import RoutingEngine

class ExecutionEngine:
    def __init__(self, db: AsyncSession):
        self.db = db
        self.context_mgr = ContextManager(db)
        self.cache_service = SemanticCacheService(db)
        self.router = RoutingEngine(db)  # Add router
    
    async def execute_workflow(self, workflow_id: uuid.UUID,
                                default_provider: str = "openai",
                                default_model: str | None = None,
                                routing_preferences: dict | None = None) -> uuid.UUID:
        """
        Execute all stages of a workflow sequentially.
        
        Args:
            workflow_id: Workflow ID
            default_provider: Fallback provider if routing fails
            default_model: Fallback model if routing fails
            routing_preferences: Optional routing preferences per stage
                Example: {"stage_1_id": "gpt-4o", "stage_2_id": "claude-3-5-sonnet"}
        
        Returns:
            execution_id
        """
        execution_id = uuid.uuid4()
        routing_preferences = routing_preferences or {}

        # Load workflow with stages
        result = await self.db.execute(
            select(Workflow).options(selectinload(Workflow.stages)).where(Workflow.id == workflow_id)
        )
        workflow = result.scalar_one_or_none()
        if not workflow:
            raise ValueError(f"Workflow {workflow_id} not found")

        if not workflow.stages:
            raise ValueError("Workflow has no stages")

        # Update workflow status
        workflow.status = "running"
        await self.db.flush()

        # Store workflow objective as context
        if workflow.objective:
            await self.context_mgr.add_context(
                workflow_id, execution_id, stage_id=None,
                content=f"Workflow Objective: {workflow.objective}",
                context_type="user_input"
            )

        # Sort stages by order
        sorted_stages = sorted(workflow.stages, key=lambda s: s.stage_order)

        all_success = True
        for stage in sorted_stages:
            try:
                # Get user override for this stage if provided
                user_override = routing_preferences.get(str(stage.id))
                
                await self._execute_stage(
                    workflow_id, execution_id, stage,
                    default_provider, default_model,
                    user_override=user_override
                )
            except Exception as e:
                all_success = False
                await self._record_failure(workflow_id, execution_id, stage, str(e))
                stage.status = "failed"
                await self.db.flush()
                break

        # Update workflow status
        workflow.status = "completed" if all_success else "failed"
        await self.db.flush()

        return execution_id

    async def _execute_stage(self, workflow_id: uuid.UUID, execution_id: uuid.UUID,
                              stage: Stage, default_provider: str,
                              default_model: str | None,
                              user_override: str | None = None) -> ExecutionRecord:
        """Execute a single stage with routing and caching."""
        stage.status = "running"
        await self.db.flush()

        # Assemble input
        stage_input = await self.context_mgr.assemble_stage_input(workflow_id, execution_id, stage)

        # ========== CACHE LOOKUP ==========
        cache_result = await self.cache_service.lookup_with_validation(
            input_text=stage_input,
            workflow_id=workflow_id,
            execution_id=execution_id,
            current_stage=stage
        )

        if cache_result.cache_hit and cache_result.entry:
            # CACHE HIT - Return cached result
            print(f"[CACHE HIT] Stage {stage.name} - Similarity: {cache_result.similarity_score:.4f}")
            
            await self.context_mgr.add_context(
                workflow_id, execution_id, stage.id,
                content=cache_result.entry.result,
                context_type="stage_output",
                token_count=cache_result.entry.result_tokens,
            )

            record = ExecutionRecord(
                workflow_id=workflow_id, stage_id=stage.id, execution_id=execution_id,
                model_used=cache_result.entry.model_used,
                provider="cache",
                input_tokens=0,
                output_tokens=cache_result.entry.result_tokens,
                latency_ms=0,
                estimated_cost=0.0,
                status="completed",
                result=cache_result.entry.result,
                started_at=datetime.now(timezone.utc),
                completed_at=datetime.now(timezone.utc),
            )
            self.db.add(record)
            stage.status = "completed"
            await self.db.flush()
            
            return record
        
        # ========== CACHE MISS - Use Router to Select Model ==========
        print(f"[CACHE MISS] Stage {stage.name} - Routing to optimal model")

        try:
            # Use routing engine to select model
            selected_model, selection_reason = await self.router.select_model_for_stage(
                stage, execution_id, user_override=user_override
            )
            provider_name = selected_model.provider
            model = selected_model.model_name
            
            print(f"[ROUTER] Selected {selected_model.display_name} - {selection_reason}")
        except Exception as e:
            # Fallback to default if routing fails
            print(f"[ROUTER ERROR] Routing failed: {e}, using default")
            provider_name = default_provider
            model = stage.model_preference or default_model

        # Get provider
        provider = registry.get(provider_name)
        if model is None:
            model = provider.get_default_model()

        # Execute with retry
        response = await self._call_with_retry(provider, stage_input, model)

        # Store result in context
        await self.context_mgr.add_context(
            workflow_id, execution_id, stage.id,
            content=response.content, context_type="stage_output",
            token_count=response.total_tokens,
        )

        # ========== STORE IN CACHE ==========
        ttl_seconds = settings.CACHE_TTL_SECONDS if settings.CACHE_TTL_SECONDS > 0 else None
        await self.cache_service.store_with_dependencies(
            input_text=stage_input,
            result=response.content,
            workflow_id=workflow_id,
            execution_id=execution_id,
            current_stage=stage,
            result_tokens=response.total_tokens,
            model_used=response.model,
            ttl_seconds=ttl_seconds
        )

        # Record execution
        record = ExecutionRecord(
            workflow_id=workflow_id, stage_id=stage.id, execution_id=execution_id,
            model_used=response.model, provider=response.provider,
            input_tokens=response.input_tokens, output_tokens=response.output_tokens,
            latency_ms=response.latency_ms, estimated_cost=response.estimated_cost,
            status="completed", result=response.content,
            started_at=datetime.now(timezone.utc),
            completed_at=datetime.now(timezone.utc),
        )
        self.db.add(record)

        stage.status = "completed"
        await self.db.flush()

        return record
```

```python
class ExecutionStartRequest(BaseModel):
    """Request to start workflow execution."""
    default_provider: Optional[str] = Field(default="openai", description="Default LLM provider")
    default_model: Optional[str] = Field(default=None, description="Default model name")
    user_inputs: Optional[dict] = Field(default_factory=dict, description="Additional user inputs")
    routing_preferences: Optional[dict] = Field(
        default_factory=dict,
        description="Model preferences per stage ID (stage_id: model_name)"
    )
```

### 4.3 Update `backend/app/api/execution.py`

Pass routing preferences to execution engine:

```python
@router.post("/workflows/{workflow_id}/execute", response_model=ExecutionStartResponse)
async def start_workflow_execution(
    workflow_id: uuid.UUID,
    request: ExecutionStartRequest,
    db: AsyncSession = Depends(get_db)
):
    """Start execution of a workflow."""
    # Verify workflow exists
    result = await db.execute(select(Workflow).where(Workflow.id == workflow_id))
    workflow = result.scalar_one_or_none()
    if not workflow:
        raise HTTPException(status_code=404, detail=f"Workflow {workflow_id} not found")

    if workflow.status == "running":
        raise HTTPException(status_code=409, detail="Workflow is already running")

    # Start execution with routing preferences
    try:
        engine = ExecutionEngine(db)
        execution_id = await engine.execute_workflow(
            workflow_id,
            default_provider=request.default_provider,
            default_model=request.default_model,
            routing_preferences=request.routing_preferences  # Pass routing prefs
        )
        await db.commit()

        return ExecutionStartResponse(
            execution_id=execution_id,
            workflow_id=workflow_id,
            status="completed",
            message="Workflow execution completed successfully",
            started_at=workflow.updated_at
        )
    except Exception as e:
        await db.rollback()
        raise HTTPException(status_code=500, detail=f"Execution failed: {str(e)}")
```
