# MT-16 — Frontend Execution UI

## 1. Context
With the execution API complete (MT-15), we now need a frontend interface to trigger workflow executions, monitor progress, and view results. This includes an "Execute" button on the workflow detail page, real-time status indicators, and a tabbed interface to view stage outputs, context flow, and execution metrics.

## 2. Prerequisites
| What | Why |
|------|-----|
| **MT-15 complete** | Execution API available |
| **MT-09 complete** | Workflow detail page exists |
| **MT-08 complete** | Frontend scaffold with routing |

## 3. Goal
- Execute button on workflow detail page.
- Execution configuration modal (select provider/model).
- Real-time execution status display (pending → running → completed/failed).
- Tabbed results view: **Stages**, **Context Flow**, **Metrics**.
- Stage output cards with token/cost/latency details.
- Context flow visualization showing how data flows between stages.
- Execution metrics summary (total tokens, cost, duration).
- Execution history list with ability to view past runs.

### 4.2 Create `frontend/src/api/executionApi.ts`

```typescript
/**
 * API client for execution operations.
 */
import { apiClient } from './client';
import type {
  ExecutionStartRequest,
  ExecutionStartResponse,
  ExecutionDetailResponse,
  ExecutionListItem
} from '../types/execution';

export const executionApi = {
  /**
   * Start workflow execution.
   */
  async startExecution(
    workflowId: string,
    request: ExecutionStartRequest
  ): Promise<ExecutionStartResponse> {
    const response = await apiClient.post(
      `/api/workflows/${workflowId}/execute`,
      request
    );
    return response.data;
  },

  /**
   * Get execution details by ID.
   */
  async getExecution(executionId: string): Promise<ExecutionDetailResponse> {
    const response = await apiClient.get(`/api/executions/${executionId}`);
    return response.data;
  },

  /**
   * List all executions, optionally filtered by workflow.
   */
  async listExecutions(
    workflowId?: string,
    limit: number = 20,
    offset: number = 0
  ): Promise<ExecutionListItem[]> {
    const params: any = { limit, offset };
    if (workflowId) params.workflow_id = workflowId;
    const response = await apiClient.get('/api/executions', { params });
    return response.data;
  },

  /**
   * Get latest execution for a workflow.
   */
  async getLatestExecution(workflowId: string): Promise<ExecutionDetailResponse> {
    const response = await apiClient.get(
      `/api/workflows/${workflowId}/executions/latest`
    );
    return response.data;
  }
};
```

/**

import React, { useState } from 'react';
import type { ExecutionDetailResponse, StageExecutionDetail } from '../types/execution';

interface ExecutionResultsProps {
  execution: ExecutionDetailResponse;
}

const STATUS_COLORS = {
  pending: 'bg-gray-100 text-gray-700',
  running: 'bg-blue-100 text-blue-700',
  completed: 'bg-green-100 text-green-700',
  failed: 'bg-red-100 text-red-700',
  skipped: 'bg-yellow-100 text-yellow-700'
};

export const ExecutionResults: React.FC<ExecutionResultsProps> = ({ execution }) => {
  const [activeTab, setActiveTab] = useState<'stages' | 'context' | 'metrics'>('stages');

  const tabs = [
    { id: 'stages', label: 'Stage Results' },
    { id: 'context', label: 'Context Flow' },
    { id: 'metrics', label: 'Metrics' }

  return (
    <div className="bg-white rounded-lg shadow-md p-6">
      {/* Header */}
      <div className="mb-6">
        <div className="flex items-center justify-between mb-2">
          <h2 className="text-2xl font-bold">Execution Results</h2>

          </span>
        </div>
        <p className="text-gray-600 text-sm">
          Execution ID: <code className="bg-gray-100 px-2 py-1 rounded">{execution.summary.execution_id}</code>
        </p>
      </div>

      {/* Tabs */}
      <div className="border-b border-gray-200 mb-6">

      {activeTab === 'context' && <ContextTab stages={execution.stages} />}

};

// --- Stage Results Tab ---
const StagesTab: React.FC<{ stages: StageExecutionDetail[] }> = ({ stages }) => {

const ContextTab: React.FC<{ stages: StageExecutionDetail[] }> = ({ stages }) => {

            </div>
          </div>

// --- Metrics Tab ---
const MetricsTab: React.FC<{

  stages: StageExecutionDetail[];

  const avgLatency = summary.total_stages > 0 ? summary.total_latency_ms / summary.total_stages : 0;
  const avgTokens = summary.total_stages > 0 ? summary.total_tokens / summary.total_stages : 0;

  return (
    <div className="space-y-6">
      {/* Summary cards */}
      <div className="grid grid-cols-2 md:grid-cols-4 gap-4">

          value={`${(summary.duration_ms || 0).toLocaleString()}ms`}

          value={summary.total_tokens.toLocaleString()}

          label="Total Cost"
          value={`$${summary.total_cost.toFixed(4)}`}

          value={`${summary.completed_stages}/${summary.total_stages}`}

              <tr>
                <th className="px-4 py-2 text-left">#</th>
                <th className="px-4 py-2 text-left">Stage</th>
                <th className="px-4 py-2 text-left">Model</th>
                <th className="px-4 py-2 text-right">Tokens</th>
                <th className="px-4 py-2 text-right">Latency</th>

### 4.5 Update `frontend/src/pages/WorkflowDetail.tsx`

Add execution functionality to existing workflow detail page:

```tsx
// Add imports
import { useState } from 'react';
import { executionApi } from '../api/executionApi';
import { ExecutionConfigModal } from '../components/ExecutionConfigModal';
import { ExecutionResults } from '../components/ExecutionResults';
import type { ExecutionDetailResponse } from '../types/execution';

// Inside component:
const [showExecuteModal, setShowExecuteModal] = useState(false);
const [execution, setExecution] = useState<ExecutionDetailResponse | null>(null);
const [isExecuting, setIsExecuting] = useState(false);

// Load latest execution on mount
useEffect(() => {
  const loadLatestExecution = async () => {
    try {
      const latest = await executionApi.getLatestExecution(workflowId);
      setExecution(latest);
    } catch (err) {
      // No executions yet, that's fine
    }
  };
  if (workflow) {
    loadLatestExecution();
  }
}, [workflow]);

const handleExecute = async (config: any) => {
  setIsExecuting(true);
  try {
    const response = await executionApi.startExecution(workflowId, config);
    const details = await executionApi.getExecution(response.execution_id);
    setExecution(details);
  } catch (error: any) {
    alert(`Execution failed: ${error.response?.data?.detail || error.message}`);
  } finally {
    setIsExecuting(false);
  }
};

// Add execute button in the UI (after workflow name header):
<div className="flex justify-between items-center mb-6">
  <h1 className="text-3xl font-bold">{workflow.name}</h1>
  <button
    onClick={() => setShowExecuteModal(true)}
    disabled={isExecuting}
    className="px-4 py-2 bg-green-600 text-white rounded-md hover:bg-green-700 disabled:opacity-50"
  >
    {isExecuting ? 'Executing...' : 'Execute Workflow'}
  </button>
</div>

// Add modal and results components at bottom of return:
<ExecutionConfigModal
  isOpen={showExecuteModal}
  onClose={() => setShowExecuteModal(false)}
  onExecute={handleExecute}
  workflowName={workflow.name}
/>

{execution && <ExecutionResults execution={execution} />}
```
