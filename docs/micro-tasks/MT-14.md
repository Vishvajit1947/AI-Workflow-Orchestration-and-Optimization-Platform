# MT-14 — Sequential Execution Engine

## 1. Context
This is the core runtime: the Execution Engine takes a workflow, resolves stage order, calls the LLM for each stage sequentially, manages context between stages, records execution metrics, and updates workflow/stage status. No caching or routing yet — it uses the stage's `model_preference` or a default model.

## 2. Prerequisites
| What | Why |
|------|-----|
| **MT-11 complete** | Context manager + execution record model |
| **MT-12 complete** | LLM providers available |
| **MT-07 complete** | Stages have dependencies |

## 3. Goal
- `backend/app/services/execution/execution_engine.py` — Run all stages sequentially.
- Records `ExecutionRecord` for each stage with tokens, latency, cost.
- Updates `Workflow.status` and `Stage.status` during execution.
- Stores stage outputs in `WorkflowContext`.
- Basic retry on failure (configurable attempts).

## 4. Implementation

### 4.1 `backend/app/services/execution/execution_engine.py`

```python
"""
Sequential Execution Engine.
Runs workflow stages in stage_order, calls LLM, stores results in context and execution records.
"""
import uuid
import time
import traceback
from datetime import datetime, timezone

from sqlalchemy import select, update
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy.orm import selectinload

from backend.app.config import settings
from backend.app.models.workflow import Workflow
from backend.app.models.stage import Stage
from backend.app.models.execution import ExecutionRecord
from backend.app.services.context_manager import ContextManager
from backend.app.services.llm import registry
from backend.app.services.llm.base import LLMResponse


class ExecutionEngine:
    def __init__(self, db: AsyncSession):
        self.db = db
        self.context_mgr = ContextManager(db)

    async def execute_workflow(self, workflow_id: uuid.UUID,
                                default_provider: str = "openai",
                                default_model: str | None = None) -> uuid.UUID:
        """Execute all stages of a workflow sequentially. Returns execution_id."""
        execution_id = uuid.uuid4()

        # Load workflow with stages
        result = await self.db.execute(
            select(Workflow).options(selectinload(Workflow.stages)).where(Workflow.id == workflow_id)
        )
        workflow = result.scalar_one_or_none()
        if not workflow:
            raise ValueError(f"Workflow {workflow_id} not found")

        if not workflow.stages:
            raise ValueError("Workflow has no stages")

        # Update workflow status
        workflow.status = "running"
        await self.db.flush()

        # Store workflow objective as context
        if workflow.objective:
            await self.context_mgr.add_context(
                workflow_id, execution_id, stage_id=None,
                content=f"Workflow Objective: {workflow.objective}",
                context_type="user_input"
            )

        # Sort stages by order
        sorted_stages = sorted(workflow.stages, key=lambda s: s.stage_order)

        all_success = True
        for stage in sorted_stages:
            try:
                await self._execute_stage(
                    workflow_id, execution_id, stage,
                    default_provider, default_model
                )
            except Exception as e:
                all_success = False
                # Record failure
                await self._record_failure(workflow_id, execution_id, stage, str(e))
                stage.status = "failed"
                await self.db.flush()
                # Stop execution on failure (sequential mode)
                break

        # Update workflow status
        workflow.status = "completed" if all_success else "failed"
        await self.db.flush()

        return execution_id

    async def _execute_stage(self, workflow_id: uuid.UUID, execution_id: uuid.UUID,
                              stage: Stage, default_provider: str,
                              default_model: str | None) -> ExecutionRecord:
        """Execute a single stage."""
        stage.status = "running"
        await self.db.flush()

        # Assemble input
        stage_input = await self.context_mgr.assemble_stage_input(workflow_id, execution_id, stage)

        # Determine provider and model
        provider_name = default_provider
        model = stage.model_preference or default_model

        # Get provider
        provider = registry.get(provider_name)
        if model is None:
            model = provider.get_default_model()

        # Execute with retry
        response = await self._call_with_retry(provider, stage_input, model)

        # Store result in context
        await self.context_mgr.add_context(
            workflow_id, execution_id, stage.id,
            content=response.content, context_type="stage_output",
            token_count=response.total_tokens,
        )

        # Record execution
        record = ExecutionRecord(
            workflow_id=workflow_id, stage_id=stage.id, execution_id=execution_id,
            model_used=response.model, provider=response.provider,
            input_tokens=response.input_tokens, output_tokens=response.output_tokens,
            latency_ms=response.latency_ms, estimated_cost=response.estimated_cost,
            status="completed", result=response.content,
            started_at=datetime.now(timezone.utc),
            completed_at=datetime.now(timezone.utc),
        )
        self.db.add(record)

        # Update stage status
        stage.status = "completed"
        await self.db.flush()

        return record

    async def _call_with_retry(self, provider, prompt: str, model: str,
                                max_retries: int | None = None) -> LLMResponse:
        """Call LLM with retry and exponential backoff."""
        import asyncio
        retries = max_retries or settings.MAX_RETRIES
        last_error = None
        for attempt in range(retries):
            try:
                return await provider.generate(
                    prompt=prompt, model=model,
                    system_prompt="You are an AI assistant helping with a multi-stage workflow. Provide detailed, actionable output for this stage."
                )
            except Exception as e:
                last_error = e
                if attempt < retries - 1:
                    wait = settings.RETRY_BACKOFF_BASE ** attempt
                    await asyncio.sleep(wait)
        raise last_error

    async def _record_failure(self, workflow_id: uuid.UUID, execution_id: uuid.UUID,
                               stage: Stage, error_msg: str):
        record = ExecutionRecord(
            workflow_id=workflow_id, stage_id=stage.id, execution_id=execution_id,
            status="failed", error_message=error_msg,
            started_at=datetime.now(timezone.utc),
            completed_at=datetime.now(timezone.utc),
        )
        self.db.add(record)
        await self.db.flush()
```

asyncio.run(test())
