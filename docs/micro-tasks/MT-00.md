### 5.6 `docker-compose.yml` skeleton (repo root)

Create a `docker-compose.yml` that defines all four services (PostgreSQL, Redis, backend, frontend) with proper networking. The backend and frontend `build:` directives point to their respective directories; actual `Dockerfile`s will be created in later MTs. This skeleton is valid YAML and can be used immediately once Dockerfiles exist.

**File: `docker-compose.yml`**

```yaml
# =============================================================================
# AI Workflow Orchestration Platform — Docker Compose
# =============================================================================
# Usage:
#   docker-compose up -d          # Start all services
#   docker-compose down           # Stop all services
#   docker-compose logs -f        # Follow logs
#
# Prerequisites:
#   - Docker and Docker Compose installed
#   - Copy .env.example to .env and fill in API keys
# =============================================================================

version: "3.9"

services:
  # ---------------------------------------------------------------------------
  # PostgreSQL 15 — primary database
  # ---------------------------------------------------------------------------
  postgres:
    image: postgres:15-alpine
    container_name: ai_orchestrator_postgres
    restart: unless-stopped
    ports:
      - "${POSTGRES_PORT:-5432}:5432"
    environment:
      POSTGRES_USER: ${POSTGRES_USER:-orchestrator}
      POSTGRES_PASSWORD: ${POSTGRES_PASSWORD:-orchestrator_dev}
      POSTGRES_DB: ${POSTGRES_DB:-ai_orchestrator}
    volumes:
      - postgres_data:/var/lib/postgresql/data
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U ${POSTGRES_USER:-orchestrator}"]
      interval: 10s
      timeout: 5s
      retries: 5

  # ---------------------------------------------------------------------------
  # Redis 7 — caching, vector storage, pub/sub for WebSocket
  # ---------------------------------------------------------------------------
  redis:
    image: redis:7-alpine
    container_name: ai_orchestrator_redis
    restart: unless-stopped
    ports:
      - "${REDIS_PORT:-6379}:6379"
    volumes:
      - redis_data:/data
    healthcheck:
      test: ["CMD", "redis-cli", "ping"]
      interval: 10s
      timeout: 5s
      retries: 5

  # ---------------------------------------------------------------------------
  # FastAPI Backend
  # ---------------------------------------------------------------------------
  backend:
    build:
      context: ./backend
      dockerfile: Dockerfile
    container_name: ai_orchestrator_backend
    restart: unless-stopped
    ports:
      - "${BACKEND_PORT:-8000}:8000"
    environment:
      - DATABASE_URL=postgresql+asyncpg://${POSTGRES_USER:-orchestrator}:${POSTGRES_PASSWORD:-orchestrator_dev}@postgres:5432/${POSTGRES_DB:-ai_orchestrator}
      - REDIS_URL=redis://redis:6379/0
      - OPENAI_API_KEY=${OPENAI_API_KEY:-}
      - ANTHROPIC_API_KEY=${ANTHROPIC_API_KEY:-}
      - GOOGLE_API_KEY=${GOOGLE_API_KEY:-}
      - ENVIRONMENT=${ENVIRONMENT:-development}
      - LOG_LEVEL=${LOG_LEVEL:-info}
    depends_on:
      postgres:
        condition: service_healthy
      redis:
        condition: service_healthy
    volumes:
      - ./backend:/app

  # ---------------------------------------------------------------------------
  # React + Vite Frontend
  # ---------------------------------------------------------------------------
  frontend:
    build:
      context: ./frontend
      dockerfile: Dockerfile
    container_name: ai_orchestrator_frontend
    restart: unless-stopped
    ports:
      - "${FRONTEND_PORT:-5173}:5173"
    environment:
      - VITE_API_BASE_URL=http://localhost:${BACKEND_PORT:-8000}
    depends_on:
      - backend
    volumes:
      - ./frontend:/app
      - /app/node_modules

volumes:
  postgres_data:
  redis_data:
```
