# MT-30 — Parallel Executor

## 1. Context
With DAG analysis complete (MT-29), we now implement the parallel executor that runs independent stages concurrently using asyncio. This dramatically reduces workflow execution time for complex DAGs.

## 2. Prerequisites
| What | Why |
|------|-----|
| **MT-29 complete** | DAG analyzer available |
| **MT-26 complete** | Router integration in execution |
| **MT-21 complete** | Cache integration complete |

## 3. Goal
- `backend/app/services/execution/parallel_executor.py` — Async parallel execution.
- Execute stages level-by-level (wait for level N before starting N+1).
- Use asyncio for concurrent stage execution within a level.
- Track stage status in real-time (pending, running, completed, failed).
- Handle partial failures (continue with independent stages).
- Merge point synchronization (wait for all dependencies).
- Update execution engine to support both sequential and parallel modes.

```python
"""
Parallel Executor.
Executes workflow stages in parallel based on DAG analysis.
"""
import asyncio
import uuid
from datetime import datetime, timezone
from typing import Dict, Set, List

from sqlalchemy.ext.asyncio import AsyncSession

from backend.app.models.workflow import Workflow
from backend.app.models.stage import Stage
from backend.app.models.execution import ExecutionRecord
from backend.app.services.execution.dag_analyzer import DAGAnalyzer
from backend.app.services.context_manager import ContextManager
from backend.app.services.cache.semantic_cache import SemanticCacheService
from backend.app.services.router.routing_engine import RoutingEngine
from backend.app.services.llm import registry


class StageExecutionState:
    """Tracks execution state of a single stage."""
    
    def __init__(self, stage: Stage):
        self.stage = stage
        self.status = "pending"  # pending, running, completed, failed, skipped
        self.result: str | None = None
        self.error: str | None = None
        self.started_at: datetime | None = None
        self.completed_at: datetime | None = None


class ParallelExecutor:
    """Executes workflow stages in parallel based on dependency graph."""
    
    def __init__(self, db: AsyncSession):
        self.db = db
        self.context_mgr = ContextManager(db)
        self.cache_service = SemanticCacheService(db)
        self.router = RoutingEngine(db)
        
        self.stage_states: Dict[uuid.UUID, StageExecutionState] = {}
        self.execution_lock = asyncio.Lock()
    
    async def execute_workflow_parallel(
        self,
        workflow: Workflow,
        execution_id: uuid.UUID,
        default_provider: str = "openai",
        default_model: str | None = None,
        routing_preferences: dict | None = None
    ) -> bool:
        """
        Execute workflow with parallel stage execution.
        
        Args:
            workflow: Workflow to execute
            execution_id: Execution ID
            default_provider: Fallback provider
            default_model: Fallback model
            routing_preferences: Per-stage model overrides
            
        Returns:
            True if all stages completed successfully
        """
        routing_preferences = routing_preferences or {}
        
        # Store workflow objective
        if workflow.objective:
            await self.context_mgr.add_context(
                workflow.id, execution_id, stage_id=None,
                content=f"Workflow Objective: {workflow.objective}",
                context_type="user_input"
            )
        
        # Analyze DAG
        analyzer = DAGAnalyzer(workflow.stages)
        execution_order = analyzer.get_execution_order()
        
        print(f"\n[PARALLEL EXECUTION] Starting workflow: {workflow.name}")
        print(analyzer.visualize_dag())
        
        # Initialize stage states
        for stage in workflow.stages:
            self.stage_states[stage.id] = StageExecutionState(stage)
        
        # Execute level by level
        all_success = True
        for level_idx, level_stages in enumerate(execution_order):
            print(f"\n[LEVEL {level_idx}] Executing {len(level_stages)} stages in parallel")
            
            # Execute all stages in this level concurrently
            tasks = [
                self._execute_stage_async(
                    workflow.id, execution_id, stage,
                    default_provider, default_model,
                    routing_preferences.get(str(stage.id))
                )
                for stage in level_stages
            ]
            
            results = await asyncio.gather(*tasks, return_exceptions=True)
            
            # Check results
            for stage, result in zip(level_stages, results):
                state = self.stage_states[stage.id]
                if isinstance(result, Exception) or state.status == "failed":
                    all_success = False
                    print(f"  ✗ {stage.name}: FAILED - {state.error}")
                else:
                    print(f"  ✓ {stage.name}: SUCCESS")
            
            # If level failed and we have dependent stages, mark them as skipped
            if not all_success:
                await self._mark_dependent_stages_skipped(analyzer, level_stages)
                break
        
        return all_success
    
    async def _execute_stage_async(
        self,
        workflow_id: uuid.UUID,
        execution_id: uuid.UUID,
        stage: Stage,
        default_provider: str,
        default_model: str | None,
        user_override: str | None = None
    ):
        """Execute a single stage asynchronously."""
        state = self.stage_states[stage.id]
        
        try:
            state.status = "running"
            state.started_at = datetime.now(timezone.utc)
            stage.status = "running"
            
            async with self.execution_lock:
                await self.db.flush()
            
            # Assemble input
            stage_input = await self.context_mgr.assemble_stage_input(
                workflow_id, execution_id, stage
            )
            
            # Cache lookup
            cache_result = await self.cache_service.lookup_with_validation(
                input_text=stage_input,
                workflow_id=workflow_id,
                execution_id=execution_id,
                current_stage=stage
            )
            
            if cache_result.cache_hit and cache_result.entry:
                # Cache hit
                print(f"  [CACHE HIT] {stage.name}")
                
                await self.context_mgr.add_context(
                    workflow_id, execution_id, stage.id,
                    content=cache_result.entry.result,
                    context_type="stage_output",
                    token_count=cache_result.entry.result_tokens,
                )
                
                record = ExecutionRecord(
                    workflow_id=workflow_id, stage_id=stage.id, execution_id=execution_id,
                    model_used=cache_result.entry.model_used,
                    provider="cache",
                    input_tokens=0,
                    output_tokens=cache_result.entry.result_tokens,
                    latency_ms=0,
                    estimated_cost=0.0,
                    status="completed",
                    result=cache_result.entry.result,
                    started_at=state.started_at,
                    completed_at=datetime.now(timezone.utc),
                )
                
                async with self.execution_lock:
                    self.db.add(record)
                    await self.db.flush()
                
                state.result = cache_result.entry.result
                state.status = "completed"
                state.completed_at = datetime.now(timezone.utc)
                stage.status = "completed"
                
                async with self.execution_lock:
                    await self.db.flush()
                
                return
            
            # Cache miss - route and execute
            print(f"  [EXECUTING] {stage.name}")
            
            selected_model, selection_reason = await self.router.select_model_for_stage(
                stage, execution_id, user_override=user_override
            )
            provider_name = selected_model.provider
            model = selected_model.model_name
            
            # Get provider
            provider = registry.get(provider_name)
            
            # Execute LLM call
            response = await provider.generate(
                prompt=stage_input,
                model=model,
                system_prompt="You are an AI assistant helping with a multi-stage workflow. Provide detailed, actionable output for this stage."
            )
            
            # Store result
            await self.context_mgr.add_context(
                workflow_id, execution_id, stage.id,
                content=response.content,
                context_type="stage_output",
                token_count=response.total_tokens,
            )
            
            # Cache result
            await self.cache_service.store_with_dependencies(
                input_text=stage_input,
                result=response.content,
                workflow_id=workflow_id,
                execution_id=execution_id,
                current_stage=stage,
                result_tokens=response.total_tokens,
                model_used=response.model,
                ttl_seconds=86400
            )
            
            # Record execution
            record = ExecutionRecord(
                workflow_id=workflow_id, stage_id=stage.id, execution_id=execution_id,
                model_used=response.model, provider=response.provider,
                input_tokens=response.input_tokens, output_tokens=response.output_tokens,
                latency_ms=response.latency_ms, estimated_cost=response.estimated_cost,
                status="completed", result=response.content,
                started_at=state.started_at,
                completed_at=datetime.now(timezone.utc),
            )
            
            async with self.execution_lock:
                self.db.add(record)
                await self.db.flush()
            
            state.result = response.content
            state.status = "completed"
            state.completed_at = datetime.now(timezone.utc)
            stage.status = "completed"
            
            async with self.execution_lock:
                await self.db.flush()
        
        except Exception as e:
            state.status = "failed"
            state.error = str(e)
            state.completed_at = datetime.now(timezone.utc)
            stage.status = "failed"
            
            # Record failure
            record = ExecutionRecord(
                workflow_id=workflow_id, stage_id=stage.id, execution_id=execution_id,
                status="failed", error_message=str(e),
                started_at=state.started_at,
                completed_at=datetime.now(timezone.utc),
            )
            
            async with self.execution_lock:
                self.db.add(record)
                stage.status = "failed"
                await self.db.flush()
            
            raise
    
    async def _mark_dependent_stages_skipped(
        self,
        analyzer: DAGAnalyzer,
        failed_level_stages: List[Stage]
    ):
        """Mark stages that depend on failed stages as skipped."""
        failed_ids = {s.id for s in failed_level_stages 
                     if self.stage_states[s.id].status == "failed"}
        
        for stage_id, state in self.stage_states.items():
            if state.status == "pending":
                # Check if this stage depends on any failed stage
                deps = analyzer.get_dependencies(stage_id)
                if any(d.id in failed_ids for d in deps):
                    state.status = "skipped"
                    state.stage.status = "skipped"
```

class ExecutionEngine:
    def __init__(self, db: AsyncSession):
        self.db = db
        self.context_mgr = ContextManager(db)
        self.cache_service = SemanticCacheService(db)
        self.router = RoutingEngine(db)

    async def execute_workflow(
        self,
        workflow_id: uuid.UUID,
        default_provider: str = "openai",
        default_model: str | None = None,
        routing_preferences: dict | None = None,
        parallel: bool = True  # Enable parallel execution by default
    ) -> uuid.UUID:
        """
        Execute workflow (sequential or parallel).
        
        Args:
            workflow_id: Workflow ID
            default_provider: Fallback provider
            default_model: Fallback model
            routing_preferences: Per-stage model overrides
            parallel: Use parallel execution if True
            
        Returns:
            execution_id
        """
        execution_id = uuid.uuid4()
        
        # Load workflow
        result = await self.db.execute(
            select(Workflow).options(selectinload(Workflow.stages))
            .where(Workflow.id == workflow_id)
        )
        workflow = result.scalar_one_or_none()
        if not workflow:
            raise ValueError(f"Workflow {workflow_id} not found")
        
        if not workflow.stages:
            raise ValueError("Workflow has no stages")
        
        workflow.status = "running"
        await self.db.flush()
        
        try:
            if parallel:
                # Use parallel executor
                all_success = await self.parallel_executor.execute_workflow_parallel(
                    workflow, execution_id,
                    default_provider, default_model, routing_preferences
                )
            else:
                # Use sequential executor (existing logic)
                all_success = await self._execute_sequential(
                    workflow, execution_id,
                    default_provider, default_model, routing_preferences
                )
            
            workflow.status = "completed" if all_success else "failed"
            await self.db.flush()
        except Exception as e:
            workflow.status = "failed"
            await self.db.flush()
            raise
        
        return execution_id

"""
Parallel Executor Tests.
"""
import pytest
import uuid
import asyncio
from unittest.mock import AsyncMock, patch
from sqlalchemy.ext.asyncio import AsyncSession

from backend.app.models.workflow import Workflow
from backend.app.models.stage import Stage
from backend.app.services.execution.parallel_executor import ParallelExecutor


@pytest.mark.asyncio
@patch("backend.app.services.llm.openai_provider.OpenAIProvider.generate")
async def test_parallel_execution_faster_than_sequential(
    mock_generate: AsyncMock,
    db_session: AsyncSession,
    mock_llm_response
):
    """Test that parallel execution is faster for independent stages."""
    import time
    
    # Create workflow with 3 parallel stages (all order 0 = no dependencies)
    workflow = Workflow(name="Parallel Test", objective="Test speed")
    db_session.add(workflow)
    await db_session.flush()
    
    stages = [
        Stage(workflow_id=workflow.id, name=f"Stage {i}",
              instruction=f"Task {i}", stage_order=0, stage_type="analysis")
        for i in range(3)
    ]
    db_session.add_all(stages)
    await db_session.commit()
    
    # Mock with delay to simulate LLM latency
    async def slow_generate(*args, **kwargs):
        await asyncio.sleep(0.5)  # 500ms delay
        return mock_llm_response(f"Result")
    
    mock_generate.side_effect = slow_generate
    
    # Execute in parallel
    executor = ParallelExecutor(db_session)
    start = time.time()
    success = await executor.execute_workflow_parallel(
        workflow, uuid.uuid4()
    )
    duration = time.time() - start
    
    await db_session.commit()
    
    assert success is True
    # Should take ~500ms (parallel), not ~1500ms (sequential)
    assert duration < 1.0
    print(f"Parallel execution: {duration:.2f}s")


@pytest.mark.asyncio
@patch("backend.app.services.llm.openai_provider.OpenAIProvider.generate")
async def test_partial_failure_continues_independent_stages(
    mock_generate: AsyncMock,
    db_session: AsyncSession,
    mock_llm_response
):
    """Test that independent stages continue even if one fails."""
    workflow = Workflow(name="Partial Failure Test")
    db_session.add(workflow)
    await db_session.flush()
    
    # 3 independent stages
    stages = [
        Stage(workflow_id=workflow.id, name=f"Stage {i}",
              instruction=f"Task {i}", stage_order=0, stage_type="analysis")
        for i in range(3)
    ]
    db_session.add_all(stages)
    await db_session.commit()
    
    # First fails, others succeed
    mock_generate.side_effect = [
        Exception("Stage 0 failed"),
        mock_llm_response("Result 1"),
        mock_llm_response("Result 2")
    ]
    
    executor = ParallelExecutor(db_session)
    success = await executor.execute_workflow_parallel(
        workflow, uuid.uuid4()
    )
    
    await db_session.commit()
    
    # Overall fails due to one failure
    assert success is False
    
    # But other stages should complete
    states = executor.stage_states
    completed_count = sum(1 for s in states.values() if s.status == "completed")
    assert completed_count >= 2  # At least 2 should complete
