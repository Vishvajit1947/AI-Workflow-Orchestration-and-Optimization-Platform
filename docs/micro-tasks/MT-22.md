# MT-22 — Frontend Cache UI

## 1. Context
With cache integration complete (MT-21), we need frontend UI to visualize cache performance, manage cache entries, and show cache hit/miss indicators in execution results. This makes cache behavior transparent to users and allows cache management.

## 2. Prerequisites
| What | Why |
|------|-----|
| **MT-21 complete** | Cache API endpoints available |
| **MT-16 complete** | Execution results UI exists |
| **MT-08 complete** | Frontend scaffold with routing |

## 3. Goal
- Cache indicators on execution results (green badge for cache hits).
- Cache statistics dashboard page.
- Cache entries list with search/filter.
- Cache management controls (clear, delete entries).
- Real-time cache hit rate chart.
- Cache performance metrics (tokens saved, cost saved).

### 4.2 Create `frontend/src/api/cacheApi.ts`

```typescript
/**
 * API client for cache operations.
 */
import { apiClient } from './client';
import type { CacheEntry, CacheStats, CacheHitRate } from '../types/cache';

export const cacheApi = {
  /**
   * List cache entries.
   */
  async listEntries(params?: {
    workflow_id?: string;
    stage_type?: string;
    is_valid?: boolean;
    limit?: number;
    offset?: number;
  }): Promise<CacheEntry[]> {
    const response = await apiClient.get('/api/cache/entries', { params });
    return response.data;
  },

  /**
   * Get cache statistics.
   */
  async getStats(workflow_id?: string): Promise<CacheStats> {
    const params = workflow_id ? { workflow_id } : {};
    const response = await apiClient.get('/api/cache/stats', { params });
    return response.data;
  },

  /**
   * Delete a cache entry.
   */
  async deleteEntry(entryId: string): Promise<void> {
    await apiClient.delete(`/api/cache/entries/${entryId}`);
  },

  /**
   * Clear cache.
   */
  async clearCache(params?: {
    workflow_id?: string;
    stage_type?: string;
  }): Promise<{ entries_invalidated: number }> {
    const response = await apiClient.post('/api/cache/clear', null, { params });
    return response.data;
  },

  /**
   * Clean up expired entries.
   */
  async cleanupExpired(): Promise<{ entries_cleaned: number }> {
    const response = await apiClient.post('/api/cache/cleanup-expired');
    return response.data;
  },

  /**
   * Get workflow cache hit rate.
   */
  async getWorkflowHitRate(workflowId: string): Promise<CacheHitRate> {
    const response = await apiClient.get(`/api/cache/hit-rate/${workflowId}`);
    return response.data;
  }
};
```

### 4.4 Create `frontend/src/pages/CacheDashboard.tsx`

```tsx
/**
 * Cache Dashboard page.
 */
import React, { useState, useEffect } from 'react';
import { cacheApi } from '../api/cacheApi';
import type { CacheStats, CacheEntry } from '../types/cache';

export const CacheDashboard: React.FC = () => {
  const [stats, setStats] = useState<CacheStats | null>(null);
  const [entries, setEntries] = useState<CacheEntry[]>([]);
  const [loading, setLoading] = useState(true);
  const [filter, setFilter] = useState<{
    stage_type?: string;
    is_valid?: boolean;
  }>({});

  useEffect(() => {
    loadData();
  }, [filter]);

  const loadData = async () => {
    setLoading(true);
    try {
      const [statsData, entriesData] = await Promise.all([
        cacheApi.getStats(),
        cacheApi.listEntries({ ...filter, limit: 50 })
      ]);
      setStats(statsData);
      setEntries(entriesData);
    } catch (error) {
      console.error('Failed to load cache data:', error);
    } finally {
      setLoading(false);
    }
  };

  const handleClearCache = async () => {
    if (!confirm('Are you sure you want to clear the entire cache?')) return;
    
    try {
      const result = await cacheApi.clearCache();
      alert(`Cleared ${result.entries_invalidated} cache entries`);
      loadData();
    } catch (error: any) {
      alert(`Failed to clear cache: ${error.message}`);
    }
  };

  const handleCleanupExpired = async () => {
    try {
      const result = await cacheApi.cleanupExpired();
      alert(`Cleaned up ${result.entries_cleaned} expired entries`);
      loadData();
    } catch (error: any) {
      alert(`Failed to cleanup: ${error.message}`);
    }
  };

  const handleDeleteEntry = async (entryId: string) => {
    if (!confirm('Delete this cache entry?')) return;
    
    try {
      await cacheApi.deleteEntry(entryId);
      alert('Cache entry deleted');
      loadData();
    } catch (error: any) {
      alert(`Failed to delete: ${error.message}`);
    }
  };

  if (loading) {
    return <div className="p-8">Loading cache data...</div>;
  }

  return (
    <div className="p-8">
      <div className="flex justify-between items-center mb-6">
        <h1 className="text-3xl font-bold">Semantic Cache Dashboard</h1>
        <div className="flex gap-3">
          <button
            onClick={handleCleanupExpired}
            className="px-4 py-2 bg-yellow-600 text-white rounded-md hover:bg-yellow-700"
          >
            Cleanup Expired
          </button>
          <button
            onClick={handleClearCache}
            className="px-4 py-2 bg-red-600 text-white rounded-md hover:bg-red-700"
          >
            Clear All Cache
          </button>
        </div>
      </div>

      {/* Stats Cards */}
      {stats && (
        <div className="grid grid-cols-2 md:grid-cols-4 gap-6 mb-8">
          <StatCard
            label="Total Cache Entries"
            value={stats.total_entries.toLocaleString()}
            icon="📦"
          />
          <StatCard
            label="Total Cache Hits"
            value={stats.total_hits.toLocaleString()}
            icon="✅"
          />
          <StatCard
            label="Cache Hit Rate"
            value={`${(stats.hit_rate * 100).toFixed(1)}%`}
            icon="📈"
          />
          <StatCard
            label="Tokens Saved"
            value={stats.total_tokens_saved.toLocaleString()}
            subtext={`$${stats.total_cost_saved.toFixed(4)} saved`}
            icon="💰"
          />
        </div>
      )}

      {/* Filters */}
      <div className="bg-white rounded-lg shadow-md p-4 mb-6">
        <h2 className="text-lg font-semibold mb-4">Filters</h2>
        <div className="flex gap-4">
          <select
            value={filter.stage_type || ''}
            onChange={(e) => setFilter({ ...filter, stage_type: e.target.value || undefined })}
            className="px-3 py-2 border border-gray-300 rounded-md"
          >
            <option value="">All Stage Types</option>
            <option value="analysis">Analysis</option>
            <option value="design">Design</option>
            <option value="generation">Generation</option>
            <option value="testing">Testing</option>
          </select>
          
          <select
            value={filter.is_valid === undefined ? '' : filter.is_valid.toString()}
            onChange={(e) => setFilter({
              ...filter,
              is_valid: e.target.value === '' ? undefined : e.target.value === 'true'
            })}
            className="px-3 py-2 border border-gray-300 rounded-md"
          >
            <option value="">All Entries</option>
            <option value="true">Valid Only</option>
            <option value="false">Invalid Only</option>
          </select>
        </div>
      </div>

      {/* Cache Entries Table */}
      <div className="bg-white rounded-lg shadow-md overflow-hidden">
        <table className="w-full">
          <thead className="bg-gray-100">
            <tr>
              <th className="px-4 py-3 text-left text-sm font-medium">Stage Type</th>
              <th className="px-4 py-3 text-left text-sm font-medium">Input Preview</th>
              <th className="px-4 py-3 text-left text-sm font-medium">Model</th>
              <th className="px-4 py-3 text-center text-sm font-medium">Hit Count</th>
              <th className="px-4 py-3 text-center text-sm font-medium">Tokens</th>
              <th className="px-4 py-3 text-center text-sm font-medium">Valid</th>
              <th className="px-4 py-3 text-center text-sm font-medium">Actions</th>
            </tr>
          </thead>
          <tbody>
            {entries.map((entry) => (
              <tr key={entry.id} className="border-b border-gray-200 hover:bg-gray-50">
                <td className="px-4 py-3 text-sm">
                  <span className="px-2 py-1 bg-blue-100 text-blue-700 rounded text-xs">
                    {entry.stage_type || 'N/A'}
                  </span>
                </td>
                <td className="px-4 py-3 text-sm text-gray-700">
                  {entry.input_text.substring(0, 60)}...
                </td>
                <td className="px-4 py-3 text-sm text-gray-600">{entry.model_used || 'N/A'}</td>
                <td className="px-4 py-3 text-sm text-center font-medium">{entry.hit_count}</td>
                <td className="px-4 py-3 text-sm text-center">{entry.result_tokens || 0}</td>
                <td className="px-4 py-3 text-center">
                  {entry.is_valid ? (
                    <span className="text-green-600">✓</span>
                  ) : (
                    <span className="text-red-600">✗</span>
                  )}
                </td>
                <td className="px-4 py-3 text-center">
                  <button
                    onClick={() => handleDeleteEntry(entry.id)}
                    className="text-red-600 hover:text-red-800 text-sm"
                  >
                    Delete
                  </button>
                </td>
              </tr>
            ))}
          </tbody>
        </table>
        
        {entries.length === 0 && (
          <div className="p-8 text-center text-gray-500">
            No cache entries found
          </div>
        )}
      </div>
    </div>
  );
};

const StatCard: React.FC<{
  label: string;
  value: string;
  subtext?: string;
  icon: string;
}> = ({ label, value, subtext, icon }) => (
  <div className="bg-white rounded-lg shadow-md p-6">
    <div className="flex items-center justify-between mb-2">
      <p className="text-sm text-gray-600">{label}</p>
      <span className="text-2xl">{icon}</span>
    </div>
    <p className="text-3xl font-bold text-gray-900">{value}</p>
    {subtext && <p className="text-sm text-gray-500 mt-1">{subtext}</p>}
  </div>
);
```
