### 4.2 Create DAG visualization component

```tsx
// frontend/src/components/DAGVisualization.tsx
import ReactFlow, { Node, Edge } from 'reactflow';
import 'reactflow/dist/style.css';

export const DAGVisualization = ({ stages, stageStatuses }) => {
  const nodes: Node[] = stages.map((stage, idx) => ({
    id: stage.id,
    data: { 
      label: stage.name,
      status: stageStatuses[stage.id] || 'pending'
    },
    position: { x: 100 + (idx % 3) * 200, y: 100 + Math.floor(idx / 3) * 100 },
    style: {
      background: getStatusColor(stageStatuses[stage.id]),
      border: '2px solid #222',
      borderRadius: '8px',
      padding: '10px'
    }
  }));
  
  const edges: Edge[] = stages.flatMap((stage, idx) => 
    idx > 0 ? [{
      id: `e${idx-1}-${idx}`,
      source: stages[idx-1].id,
      target: stage.id,
      animated: stageStatuses[stage.id] === 'running'
    }] : []
  );
  
  return (
    <div style={{ height: '500px', border: '1px solid #ddd' }}>
      <ReactFlow nodes={nodes} edges={edges} fitView />
    </div>
  );
};

const getStatusColor = (status: string) => {
  const colors = {
    pending: '#e5e7eb',
    running: '#3b82f6',
    completed: '#10b981',
    failed: '#ef4444',
    skipped: '#f59e0b'
  };
  return colors[status] || colors.pending;
};
```

### 4.3 Execution monitor page

```tsx
// frontend/src/pages/ExecutionMonitor.tsx
import { useEffect, useState } from 'react';
import { useExecutionStatus } from '../hooks/useExecutionStatus';
import { DAGVisualization } from '../components/DAGVisualization';
import { ExecutionControls } from '../components/ExecutionControls';

export const ExecutionMonitor = ({ executionId, workflowId }) => {
  const [workflow, setWorkflow] = useState(null);
  const stageStatuses = useExecutionStatus(executionId);
  
  useEffect(() => {
    // Load workflow
    fetch(`/api/workflows/${workflowId}`)
      .then(r => r.json())
      .then(setWorkflow);
  }, [workflowId]);
  
  if (!workflow) return <div>Loading...</div>;
  
  return (
    <div className="p-8">
      <div className="flex justify-between items-center mb-6">
        <h1 className="text-3xl font-bold">Live Execution</h1>
        <ExecutionControls 
          executionId={executionId} 
          status={workflow.status} 
        />
      </div>
      
      <DAGVisualization 
        stages={workflow.stages} 
        stageStatuses={stageStatuses} 
      />
      
      <div className="mt-6">
        <h2 className="text-xl font-semibold mb-4">Stage Details</h2>
        {workflow.stages.map(stage => (
          <StageStatusCard 
            key={stage.id} 
            stage={stage} 
            status={stageStatuses[stage.id]} 
          />
        ))}
      </div>
    </div>
  );
};

const StageStatusCard = ({ stage, status }) => (
  <div className="border rounded-lg p-4 mb-3">
    <div className="flex items-center gap-3">
      <StatusBadge status={status} />
      <h3 className="font-semibold">{stage.name}</h3>
    </div>
  </div>
);

const StatusBadge = ({ status }) => {
  const badges = {
    pending: '⏳ Pending',
    running: '🔄 Running...',
    completed: '✅ Completed',
    failed: '❌ Failed',
    skipped: '⏭ Skipped'
  };
  return <span>{badges[status] || badges.pending}</span>;
};
```
