# MT-29 — DAG Analyzer

## 1. Context
Phase 5 introduces parallel execution of independent workflow stages. Instead of running stages sequentially, we analyze the dependency graph (DAG) to identify stages that can run concurrently. This MT creates the DAG analyzer that builds the dependency graph and performs topological sorting.

## 2. Prerequisites
| What | Why |
|------|-----|
| **MT-07 complete** | Stage dependencies exist |
| **MT-06 complete** | Stage model with ordering |
| **MT-01 complete** | Database connection configured |

## 3. Goal
- `backend/app/services/execution/dag_analyzer.py` — DAG construction and analysis.
- Build dependency graph from workflow stages.
- Topological sort to determine execution order.
- Identify independent stages that can run in parallel.
- Detect merge points where parallel branches reconverge.
- Calculate execution levels (batches of parallel stages).
- Validate DAG (no cycles, all dependencies satisfied).

### 4.1 Create `backend/app/services/execution/dag_analyzer.py`

```python
"""
DAG Analyzer.
Analyzes workflow dependency graphs for parallel execution.
"""
import uuid
from typing import List, Dict, Set, Tuple
from collections import defaultdict, deque

from backend.app.models.stage import Stage


class DAGNode:
    """Represents a node in the DAG."""
    
    def __init__(self, stage: Stage):
        self.stage = stage
        self.stage_id = stage.id
        self.stage_order = stage.stage_order
        self.dependencies: Set[uuid.UUID] = set()  # Stages this depends on
        self.dependents: Set[uuid.UUID] = set()    # Stages that depend on this
        self.level: int = 0  # Execution level (0 = no dependencies)
    
    def __repr__(self):
        return f"<DAGNode({self.stage.name}, order={self.stage_order}, level={self.level})>"


class ExecutionLevel:
    """A batch of stages that can execute in parallel."""
    
    def __init__(self, level: int, stages: List[Stage]):
        self.level = level
        self.stages = stages
    
    def __repr__(self):
        return f"<Level {self.level}: {len(self.stages)} stages>"


class DAGAnalyzer:
    """
    Analyzes workflow dependency graphs.
    """
    
    def __init__(self, stages: List[Stage]):
        self.stages = sorted(stages, key=lambda s: s.stage_order)
        self.nodes: Dict[uuid.UUID, DAGNode] = {}
        self.execution_levels: List[ExecutionLevel] = []
        
        self._build_graph()
    
    def _build_graph(self):
        """Build the dependency graph from stages."""
        # Create nodes
        for stage in self.stages:
            self.nodes[stage.id] = DAGNode(stage)
        
        # Build dependency edges
        for stage in self.stages:
            node = self.nodes[stage.id]
            
            # Explicit dependencies from stage_dependencies table
            if hasattr(stage, 'dependencies') and stage.dependencies:
                for dep in stage.dependencies:
                    dep_stage_id = dep.depends_on_stage_id
                    if dep_stage_id in self.nodes:
                        node.dependencies.add(dep_stage_id)
                        self.nodes[dep_stage_id].dependents.add(stage.id)
            
            # Implicit sequential dependency (stage depends on all previous stages)
            # This is the default behavior when no explicit dependencies
            if not node.dependencies:
                for prev_stage in self.stages:
                    if prev_stage.stage_order < stage.stage_order:
                        node.dependencies.add(prev_stage.id)
                        self.nodes[prev_stage.id].dependents.add(stage.id)
        
        # Calculate execution levels
        self._calculate_levels()
    
    def _calculate_levels(self):
        """
        Calculate execution levels using topological sort.
        Stages at the same level can execute in parallel.
        """
        # Calculate in-degree for each node
        in_degree = {node_id: len(node.dependencies) for node_id, node in self.nodes.items()}
        
        # Queue of nodes with no dependencies (level 0)
        queue = deque([node_id for node_id, degree in in_degree.items() if degree == 0])
        
        level_map: Dict[uuid.UUID, int] = {}
        
        while queue:
            # Get all nodes at current level
            current_level_size = len(queue)
            current_level_nodes = []
            
            for _ in range(current_level_size):
                node_id = queue.popleft()
                node = self.nodes[node_id]
                
                # Determine level (max of all dependency levels + 1)
                max_dep_level = -1
                for dep_id in node.dependencies:
                    if dep_id in level_map:
                        max_dep_level = max(max_dep_level, level_map[dep_id])
                
                node.level = max_dep_level + 1
                level_map[node_id] = node.level
                current_level_nodes.append(node_id)
                
                # Reduce in-degree for dependents
                for dependent_id in node.dependents:
                    in_degree[dependent_id] -= 1
                    if in_degree[dependent_id] == 0:
                        queue.append(dependent_id)
            
            # Group nodes by level
            if current_level_nodes:
                # Find actual level (may vary due to dependencies)
                actual_level = max(self.nodes[nid].level for nid in current_level_nodes)
                level_stages = [self.nodes[nid].stage for nid in current_level_nodes 
                               if self.nodes[nid].level == actual_level]
                if level_stages:
                    self.execution_levels.append(ExecutionLevel(actual_level, level_stages))
        
        # Sort levels
        self.execution_levels.sort(key=lambda el: el.level)
        
        # Verify all nodes were processed (no cycles)
        if len(level_map) != len(self.nodes):
            raise ValueError("Cycle detected in workflow dependencies")
    
    def get_execution_order(self) -> List[List[Stage]]:
        """
        Get execution order as list of batches.
        Each batch contains stages that can run in parallel.
        
        Returns:
            List of stage batches (outer list = levels, inner list = parallel stages)
        """
        return [level.stages for level in self.execution_levels]
    
    def get_independent_stages(self) -> List[Stage]:
        """Get stages with no dependencies (can start immediately)."""
        return [node.stage for node in self.nodes.values() if not node.dependencies]
    
    def get_dependencies(self, stage_id: uuid.UUID) -> List[Stage]:
        """Get all stages that a given stage depends on."""
        if stage_id not in self.nodes:
            return []
        
        node = self.nodes[stage_id]
        return [self.nodes[dep_id].stage for dep_id in node.dependencies]
    
    def get_dependents(self, stage_id: uuid.UUID) -> List[Stage]:
        """Get all stages that depend on a given stage."""
        if stage_id not in self.nodes:
            return []
        
        node = self.nodes[stage_id]
        return [self.nodes[dep_id].stage for dep_id in node.dependents]
    
    def is_merge_point(self, stage_id: uuid.UUID) -> bool:
        """Check if a stage is a merge point (multiple dependencies)."""
        if stage_id not in self.nodes:
            return False
        return len(self.nodes[stage_id].dependencies) > 1
    
    def get_parallelism_factor(self) -> float:
        """
        Calculate parallelism factor.
        
        Returns:
            Average number of stages that can run in parallel
        """
        if not self.execution_levels:
            return 1.0
        
        total_stages = len(self.stages)
        total_levels = len(self.execution_levels)
        
        return total_stages / total_levels if total_levels > 0 else 1.0
    
    def get_critical_path(self) -> List[Stage]:
        """
        Get the critical path (longest path through the DAG).
        This determines minimum execution time.
        """
        # Find longest path using dynamic programming
        longest_path: Dict[uuid.UUID, List[uuid.UUID]] = {}
        
        # Process nodes in topological order (by level)
        for level in self.execution_levels:
            for stage in level.stages:
                node = self.nodes[stage.id]
                
                if not node.dependencies:
                    # No dependencies = start of path
                    longest_path[stage.id] = [stage.id]
                else:
                    # Find longest path from dependencies
                    max_path = []
                    for dep_id in node.dependencies:
                        if dep_id in longest_path:
                            dep_path = longest_path[dep_id]
                            if len(dep_path) > len(max_path):
                                max_path = dep_path
                    
                    longest_path[stage.id] = max_path + [stage.id]
        
        # Find the longest path overall
        if not longest_path:
            return []
        
        critical_path_ids = max(longest_path.values(), key=len)
        return [self.nodes[sid].stage for sid in critical_path_ids]
    
    def visualize_dag(self) -> str:
        """
        Generate ASCII visualization of the DAG.
        
        Returns:
            String representation of the DAG
        """
        lines = []
        lines.append("Execution DAG:")
        lines.append("=" * 60)
        
        for level in self.execution_levels:
            lines.append(f"\nLevel {level.level} (Parallel):")
            for stage in level.stages:
                node = self.nodes[stage.id]
                deps = [self.nodes[d].stage.name for d in node.dependencies]
                deps_str = f" <- {', '.join(deps)}" if deps else " (No dependencies)"
                lines.append(f"  • {stage.name}{deps_str}")
        
        lines.append("\n" + "=" * 60)
        lines.append(f"Total Stages: {len(self.stages)}")
        lines.append(f"Execution Levels: {len(self.execution_levels)}")
        lines.append(f"Parallelism Factor: {self.get_parallelism_factor():.2f}x")
        
        critical_path = self.get_critical_path()
        if critical_path:
            lines.append(f"Critical Path: {' → '.join(s.name for s in critical_path)}")
        
        return "\n".join(lines)
```

"""
DAG Analyzer Tests.

import pytest
import uuid
from backend.app.models.workflow import Workflow
from backend.app.models.stage import Stage
from backend.app.services.execution.dag_analyzer import DAGAnalyzer


def test_sequential_workflow():
    """Test DAG analysis for sequential workflow."""
    # Create 3 sequential stages
    workflow_id = uuid.uuid4()
    stages = [
        Stage(id=uuid.uuid4(), workflow_id=workflow_id, name=f"Stage {i}",
              instruction="Test", stage_order=i, stage_type="analysis")
        for i in range(3)
    ]
    
    analyzer = DAGAnalyzer(stages)
    execution_order = analyzer.get_execution_order()
    
    # Sequential workflow = 3 levels, 1 stage each
    assert len(execution_order) == 3
    assert all(len(batch) == 1 for batch in execution_order)
    
    # Parallelism factor = 1 (no parallelism)
    assert analyzer.get_parallelism_factor() == 1.0
    
    # Critical path = all 3 stages
    critical_path = analyzer.get_critical_path()
    assert len(critical_path) == 3

    """Test DAG analysis for parallel workflow."""

    stage0 = Stage(id=uuid.uuid4(), workflow_id=workflow_id, name="Root",

### 4.4 Create utility script to visualize workflow DAG

Create `backend/scripts/visualize_dag.py`:

```python
"""
Utility to visualize workflow DAG.
"""
import asyncio
import sys
import uuid

from backend.app.database import async_session_factory
from backend.app.models.workflow import Workflow
from backend.app.services.execution.dag_analyzer import DAGAnalyzer
from sqlalchemy import select
from sqlalchemy.orm import selectinload


async def visualize_workflow_dag(workflow_id: str):
    """Visualize DAG for a workflow."""
    async with async_session_factory() as db:
        result = await db.execute(
            select(Workflow)
            .options(selectinload(Workflow.stages))
            .where(Workflow.id == uuid.UUID(workflow_id))
        )
        workflow = result.scalar_one_or_none()
        
        if not workflow:
            print(f"Workflow {workflow_id} not found")
            return
        
        if not workflow.stages:
            print("Workflow has no stages")
            return
        
        print(f"\nWorkflow: {workflow.name}")
        print(f"Objective: {workflow.objective}\n")
        
        analyzer = DAGAnalyzer(workflow.stages)
        print(analyzer.visualize_dag())


if __name__ == "__main__":
    if len(sys.argv) != 2:
        print("Usage: python visualize_dag.py <workflow_id>")
        sys.exit(1)
    
    asyncio.run(visualize_workflow_dag(sys.argv[1]))
```
