### 4.1 Create `backend/app/services/analytics/aggregator.py`

```python
"""
Analytics Aggregation Service.
Computes metrics from execution data.
"""
from datetime import datetime, timedelta
from decimal import Decimal
from typing import List, Dict, Optional
from collections import defaultdict

from sqlalchemy import select, func, and_
from sqlalchemy.ext.asyncio import AsyncSession

from backend.app.models.workflow import Workflow
from backend.app.models.execution import ExecutionRecord
from backend.app.models.cache import CacheEntry
from backend.app.models.routing_rule import RoutingDecision


class AnalyticsAggregator:
    """Service for aggregating analytics data."""
    
    def __init__(self, db: AsyncSession):
        self.db = db
    
    async def get_workflow_metrics(
        self,
        workflow_id: Optional[str] = None,
        start_date: Optional[datetime] = None,
        end_date: Optional[datetime] = None
    ) -> Dict:
        """
        Get aggregated workflow metrics.
        
        Args:
            workflow_id: Filter by specific workflow
            start_date: Start of time range
            end_date: End of time range
            
        Returns:
            Dictionary with workflow metrics
        """
        # Build query
        query = select(
            func.count(ExecutionRecord.execution_id.distinct()).label('total_executions'),
            func.count(ExecutionRecord.id).label('total_stages'),
            func.sum(ExecutionRecord.input_tokens + ExecutionRecord.output_tokens).label('total_tokens'),
            func.sum(ExecutionRecord.estimated_cost).label('total_cost'),
            func.avg(ExecutionRecord.latency_ms).label('avg_latency'),
            func.sum(func.cast(ExecutionRecord.status == 'completed', Decimal)).label('completed_stages'),
            func.sum(func.cast(ExecutionRecord.status == 'failed', Decimal)).label('failed_stages'),
        )
        
        # Apply filters
        conditions = []
        if workflow_id:
            conditions.append(ExecutionRecord.workflow_id == workflow_id)
        if start_date:
            conditions.append(ExecutionRecord.created_at >= start_date)
        if end_date:
            conditions.append(ExecutionRecord.created_at <= end_date)
        
        if conditions:
            query = query.where(and_(*conditions))
        
        result = await self.db.execute(query)
        row = result.first()
        
        total_stages = row.total_stages or 0
        completed = row.completed_stages or 0
        failed = row.failed_stages or 0
        
        return {
            'total_executions': row.total_executions or 0,
            'total_stages': total_stages,
            'total_tokens': int(row.total_tokens or 0),
            'total_cost': float(row.total_cost or 0),
            'avg_latency_ms': float(row.avg_latency or 0),
            'success_rate': (completed / total_stages * 100) if total_stages > 0 else 0,
            'failure_rate': (failed / total_stages * 100) if total_stages > 0 else 0
        }
    
    async def get_stage_type_metrics(
        self,
        start_date: Optional[datetime] = None,
        end_date: Optional[datetime] = None
    ) -> List[Dict]:
        """Get metrics grouped by stage type."""
        query = select(
            ExecutionRecord.stage_id,
            func.count(ExecutionRecord.id).label('execution_count'),
            func.avg(ExecutionRecord.latency_ms).label('avg_latency'),
            func.sum(ExecutionRecord.input_tokens + ExecutionRecord.output_tokens).label('total_tokens'),
            func.sum(ExecutionRecord.estimated_cost).label('total_cost'),
        ).join(
            ExecutionRecord.stage
        ).group_by(
            ExecutionRecord.stage_id
        )
        
        # Apply date filters
        if start_date:
            query = query.where(ExecutionRecord.created_at >= start_date)
        if end_date:
            query = query.where(ExecutionRecord.created_at <= end_date)
        
        result = await self.db.execute(query)
        rows = result.all()
        
        return [
            {
                'stage_type': row[0],  # Would need to join to get actual stage_type
                'execution_count': row.execution_count,
                'avg_latency_ms': float(row.avg_latency or 0),
                'total_tokens': int(row.total_tokens or 0),
                'total_cost': float(row.total_cost or 0)
            }
            for row in rows
        ]
    
    async def get_model_utilization(
        self,
        start_date: Optional[datetime] = None,
        end_date: Optional[datetime] = None
    ) -> List[Dict]:
        """Get model usage statistics."""
        query = select(
            ExecutionRecord.provider,
            ExecutionRecord.model_used,
            func.count(ExecutionRecord.id).label('usage_count'),
            func.sum(ExecutionRecord.input_tokens + ExecutionRecord.output_tokens).label('total_tokens'),
            func.sum(ExecutionRecord.estimated_cost).label('total_cost'),
            func.avg(ExecutionRecord.latency_ms).label('avg_latency')
        ).where(
            ExecutionRecord.model_used.isnot(None)
        ).group_by(
            ExecutionRecord.provider,
            ExecutionRecord.model_used
        ).order_by(
            func.count(ExecutionRecord.id).desc()
        )
        
        if start_date:
            query = query.where(ExecutionRecord.created_at >= start_date)
        if end_date:
            query = query.where(ExecutionRecord.created_at <= end_date)
        
        result = await self.db.execute(query)
        rows = result.all()
        
        return [
            {
                'provider': row.provider,
                'model': row.model_used,
                'usage_count': row.usage_count,
                'total_tokens': int(row.total_tokens or 0),
                'total_cost': float(row.total_cost or 0),
                'avg_latency_ms': float(row.avg_latency or 0)
            }
            for row in rows
        ]
    
    async def get_cache_metrics(self) -> Dict:
        """Get cache performance metrics."""
        # Total entries
        total_query = select(func.count(CacheEntry.id)).where(CacheEntry.is_valid == True)
        total_result = await self.db.execute(total_query)
        total_entries = total_result.scalar() or 0
        
        # Total hits
        hits_query = select(func.sum(CacheEntry.hit_count)).where(CacheEntry.is_valid == True)
        hits_result = await self.db.execute(hits_query)
        total_hits = hits_result.scalar() or 0
        
        # Total tokens saved
        tokens_query = select(func.sum(CacheEntry.result_tokens * CacheEntry.hit_count)).where(
            CacheEntry.is_valid == True
        )
        tokens_result = await self.db.execute(tokens_query)
        total_tokens_saved = tokens_result.scalar() or 0
        
        # Calculate hit rate
        total_lookups = total_entries + total_hits
        hit_rate = (total_hits / total_lookups * 100) if total_lookups > 0 else 0
        
        # Estimate cost saved (simplified)
        cost_per_token = Decimal("0.000001")
        total_cost_saved = Decimal(total_tokens_saved or 0) * cost_per_token
        
        return {
            'total_entries': total_entries,
            'total_hits': total_hits,
            'hit_rate': float(hit_rate),
            'total_tokens_saved': int(total_tokens_saved or 0),
            'total_cost_saved': float(total_cost_saved)
        }
    
    async def get_cost_trend(
        self,
        days: int = 30,
        workflow_id: Optional[str] = None
    ) -> List[Dict]:
        """Get daily cost trend."""
        end_date = datetime.now()
        start_date = end_date - timedelta(days=days)
        
        query = select(
            func.date(ExecutionRecord.created_at).label('date'),
            func.sum(ExecutionRecord.estimated_cost).label('daily_cost'),
            func.count(ExecutionRecord.execution_id.distinct()).label('execution_count')
        ).where(
            ExecutionRecord.created_at >= start_date
        ).group_by(
            func.date(ExecutionRecord.created_at)
        ).order_by(
            func.date(ExecutionRecord.created_at)
        )
        
        if workflow_id:
            query = query.where(ExecutionRecord.workflow_id == workflow_id)
        
        result = await self.db.execute(query)
        rows = result.all()
        
        return [
            {
                'date': row.date.isoformat(),
                'cost': float(row.daily_cost or 0),
                'executions': row.execution_count
            }
            for row in rows
        ]
    
    async def get_token_usage_trend(self, days: int = 30) -> List[Dict]:
        """Get daily token usage trend."""
        end_date = datetime.now()
        start_date = end_date - timedelta(days=days)
        
        query = select(
            func.date(ExecutionRecord.created_at).label('date'),
            func.sum(ExecutionRecord.input_tokens).label('input_tokens'),
            func.sum(ExecutionRecord.output_tokens).label('output_tokens')
        ).where(
            ExecutionRecord.created_at >= start_date
        ).group_by(
            func.date(ExecutionRecord.created_at)
        ).order_by(
            func.date(ExecutionRecord.created_at)
        )
        
        result = await self.db.execute(query)
        rows = result.all()
        
        return [
            {
                'date': row.date.isoformat(),
                'input_tokens': int(row.input_tokens or 0),
                'output_tokens': int(row.output_tokens or 0),
                'total_tokens': int((row.input_tokens or 0) + (row.output_tokens or 0))
            }
            for row in rows
        ]
    
    async def get_routing_effectiveness(self) -> Dict:
        """Analyze routing decision effectiveness."""
        # Total routing decisions
        total_query = select(func.count(RoutingDecision.id))
        total_result = await self.db.execute(total_query)
        total_decisions = total_result.scalar() or 0
        
        # User overrides
        override_query = select(func.count(RoutingDecision.id)).where(
            RoutingDecision.was_user_override == True
        )
        override_result = await self.db.execute(override_query)
        override_count = override_result.scalar() or 0
        
        # Fallbacks
        fallback_query = select(func.count(RoutingDecision.id)).where(
            RoutingDecision.was_fallback == True
        )
        fallback_result = await self.db.execute(fallback_query)
        fallback_count = fallback_result.scalar() or 0
        
        # Priority distribution
        priority_query = select(
            RoutingDecision.priority_factor,
            func.count(RoutingDecision.id).label('count')
        ).where(
            RoutingDecision.priority_factor.isnot(None)
        ).group_by(
            RoutingDecision.priority_factor
        )
        priority_result = await self.db.execute(priority_query)
        priority_dist = {row.priority_factor: row.count for row in priority_result.all()}
        
        return {
            'total_decisions': total_decisions,
            'override_count': override_count,
            'override_rate': (override_count / total_decisions * 100) if total_decisions > 0 else 0,
            'fallback_count': fallback_count,
            'fallback_rate': (fallback_count / total_decisions * 100) if total_decisions > 0 else 0,
            'priority_distribution': priority_dist
        }
```

### 4.2 Create `backend/app/schemas/analytics.py`

```python
"""
Pydantic schemas for analytics.
"""
from datetime import datetime
from typing import List, Dict, Optional
from pydantic import BaseModel


class WorkflowMetrics(BaseModel):
    """Workflow-level metrics."""
    total_executions: int
    total_stages: int
    total_tokens: int
    total_cost: float
    avg_latency_ms: float
    success_rate: float
    failure_rate: float


class ModelUtilization(BaseModel):
    """Model usage statistics."""
    provider: str
    model: str
    usage_count: int
    total_tokens: int
    total_cost: float
    avg_latency_ms: float


class CacheMetrics(BaseModel):
    """Cache performance metrics."""
    total_entries: int
    total_hits: int
    hit_rate: float
    total_tokens_saved: int
    total_cost_saved: float


class CostTrendPoint(BaseModel):
    """Single point in cost trend."""
    date: str
    cost: float
    executions: int


class TokenTrendPoint(BaseModel):
    """Single point in token usage trend."""
    date: str
    input_tokens: int
    output_tokens: int
    total_tokens: int


class RoutingEffectiveness(BaseModel):
    """Routing decision effectiveness metrics."""
    total_decisions: int
    override_count: int
    override_rate: float
    fallback_count: int
    fallback_rate: float
    priority_distribution: Dict[str, int]
```

```python
"""
Analytics Aggregator Tests.
"""
import pytest
from datetime import datetime, timedelta
from sqlalchemy.ext.asyncio import AsyncSession

from backend.app.services.analytics.aggregator import AnalyticsAggregator
from backend.app.models.execution import ExecutionRecord
from backend.app.models.workflow import Workflow


@pytest.mark.asyncio
async def test_workflow_metrics_aggregation(db_session: AsyncSession):
    """Test workflow metrics calculation."""
    # Create test data
    workflow = Workflow(name="Test")
    db_session.add(workflow)
    await db_session.flush()
    
    # Add execution records
    for i in range(5):
        record = ExecutionRecord(
            workflow_id=workflow.id,
            execution_id=workflow.id,  # Simplified
            model_used="gpt-4o-mini",
            provider="openai",
            input_tokens=100,
            output_tokens=50,
            latency_ms=500,
            estimated_cost=0.0001,
            status="completed"

    assert metrics['success_rate'] == 100.0
```
